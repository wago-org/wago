import pathlib,tempfile,subprocess,json,os,re,hashlib
root=pathlib.Path(tempfile.mkdtemp(prefix='wago-direct-dispatch-'))
wsrc=pathlib.Path('/tmp/wago-setup-cleanup-pr-uvyppyeh/wago');psrc=pathlib.Path('/tmp/wago-setup-cleanup-pr-uvyppyeh/wasi-publish')
for repo,dest in [(wsrc,root/'wago'),(psrc,root/'provider')]:subprocess.run(['git','-C',str(repo),'worktree','add','--detach',str(dest),'HEAD'],check=True,stdout=subprocess.DEVNULL)
s=(root/'provider/internal/core/core.go').read_text()
names=re.findall(r'\{"[^"\n]+", \(\*Plugin\)\.(\w+),',s)
assert len(names)==len(set(names))==46,len(names)
c=s.replace('handler         func(*Plugin, wago.HostModule, []uint64, []uint64)','handler         handlerID').replace('handler(&current, caller, call.ParamSlots(), call.ResultSlots())','handler.call(&current, caller, call.ParamSlots(), call.ResultSlots())')
for name in names:c=c.replace('(*Plugin).'+name+',','dispatch'+name+',')
code='\ntype handlerID uint8\n\nconst (\n'+''.join('\tdispatch'+name+(' handlerID = iota' if i==0 else '')+'\n' for i,name in enumerate(names))+')\n\nfunc (h handlerID) call(e *Plugin, m wago.HostModule, p, r []uint64) {\n\tswitch h {\n'+''.join('\tcase dispatch'+name+':\n\t\te.'+name+'(m, p, r)\n' for name in names)+'\tdefault:\n\t\tpanic("invalid WASI handler")\n\t}\n}\n'
c=c.replace('\ntype guestCapability struct',code+'\ntype guestCapability struct')
(root/'core-candidate.go').write_text(c);subprocess.run(['gofmt','-w',str(root/'core-candidate.go')],check=True)
(root/'overlay.json').write_text(json.dumps({'Replace':{str(root/'provider/internal/core/core.go'):str(root/'core-candidate.go')}}))
work=root/'build.work';work.write_text('go 1.22.0\nuse (\n'+''.join(' '+json.dumps(str(p))+'\n' for p in [root/'wago',root/'wago/bench',root/'wago/cli/wago-installer',root/'provider'])+')\nreplace github.com/wago-org/wago v0.1.0-beta.9 => '+json.dumps(str(root/'wago'))+'\n')
env=dict(os.environ,GOWORK=str(work),GOMAXPROCS='16',GOGC='100',GOMEMLIMIT='off',GODEBUG='',WAGO_BOUNDS='signals')
meta={'comparison':'private direct handler dispatch','wago_commit':subprocess.check_output(['git','-C',str(wsrc),'rev-parse','HEAD'],text=True).strip(),'provider_base':subprocess.check_output(['git','-C',str(psrc),'rev-parse','HEAD'],text=True).strip(),'binaries':{},'build_invocations':[],'benchmark_hashes':{str(p.relative_to(root/'wago')):hashlib.sha256(p.read_bytes()).hexdigest() for p in (root/'wago/bench/suite').glob('*.go')}}
for label in ['baseline','candidate']:
 binary=root/f'wasi-{label}.test';cmd=['go','test','-c','-tags','wago_guardpage','-o',str(binary)]
 if label=='candidate':cmd+=['-overlay',str(root/'overlay.json')]
 cmd+=['./bench/suite']
 subprocess.run(cmd,cwd=root/'wago',env=env,check=True)
 meta['binaries'][label]={'path':str(binary),'sha256':hashlib.sha256(binary.read_bytes()).hexdigest()}
 meta['build_invocations'].append({'command':cmd,'cwd':str(root/'wago'),'environment':json.loads(subprocess.check_output(['go','env','-json'],cwd=root/'wago',env=env)),'runtime_environment':{k:env[k] for k in ['GOMAXPROCS','GOGC','GOMEMLIMIT','GODEBUG','WAGO_BOUNDS']}})
meta['source_hashes']={label:hashlib.sha256((root/('core-candidate.go' if label=='candidate' else 'provider/internal/core/core.go')).read_bytes()).hexdigest() for label in ['baseline','candidate']}
meta['dependencies']=subprocess.check_output(['go','list','-m','-json','all'],cwd=root/'wago',env=env,text=True)
(root/'build.json').write_text(json.dumps(meta,indent=2)+'\n')
pathlib.Path('/tmp/wago-direct-latest').write_text(str(root))
for label in ['baseline','candidate']:
 with (root/(label+'-pilot.txt')).open('w') as f:
  subprocess.run(['taskset','-c','0-15',str(root/f'wasi-{label}.test'),'-test.run','^$','-test.bench','^BenchmarkWASIHostCallDiagnostic$','-test.benchtime','1000x','-test.benchmem','-wago.corpus','cjson','-wago.bench.lifecycle'],cwd=root/'wago/bench/suite',env=env,stdout=f,stderr=subprocess.STDOUT,check=True)
 print(label,(root/(label+'-pilot.txt')).read_text())
print('EXPERIMENT',root)
