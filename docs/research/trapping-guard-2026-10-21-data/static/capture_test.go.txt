//go:build linux && amd64 && wago_codegenstats

package wago

import (
	"crypto/sha256"
	"encoding/hex"
	"encoding/json"
	"os"
	"testing"
	"time"
)

func TestRealTrappingGuardStaticAdmission(t *testing.T) {
	path := os.Getenv("WAGO_REAL_COMPILER_WASM")
	out := os.Getenv("WAGO_REAL_COMPILER_STATS")
	if path == "" || out == "" {
		t.Fatal("set WAGO_REAL_COMPILER_WASM and WAGO_REAL_COMPILER_STATS")
	}
	data, err := os.ReadFile(path)
	if err != nil {
		t.Fatal(err)
	}
	realTrappingGuardCapturedStats = nil
	start := time.Now()
	c, err := Compile(NewRuntimeConfig().WithCoreFeatures(CoreFeaturesV3).WithGCCodeTelemetry(true), data)
	seconds := time.Since(start).Seconds()
	if err != nil {
		t.Fatal(err)
	}
	defer c.Close()
	stats := realTrappingGuardCapturedStats
	if stats == nil {
		t.Fatal("API did not expose stats through diagnostic overlay")
	}
	roots := c.GCNativeRootAdmission()
	if !roots.Required || !roots.Exact {
		t.Fatalf("roots=%+v", roots)
	}
	total, functions := 0, 0
	rows := make([]map[string]any, 0)
	for _, s := range stats.Funcs {
		if s == nil {
			continue
		}
		n := s.Peephole["gc-trapping-guard-reuse"]
		total += n
		if n == 0 {
			continue
		}
		functions++
		rows = append(rows, map[string]any{"local_function": s.FuncIdx, "name": s.Name, "fusions": n, "code_bytes": s.CodeBytes, "frame_bytes": s.FrameBytes, "max_spill_slots": s.MaxSpillSlots, "global_pins": s.PinnedGlobalsValue, "local_pins": s.PinnedLocals, "spills": s.Spills, "rewrites": s.Peephole, "calls": s.Calls, "gc_code_bytes": s.GCCodeBytes})
	}
	wasmHash := sha256.Sum256(data)
	codeHash := sha256.New()
	if _, err = c.WriteCodeTo(codeHash); err != nil {
		t.Fatal(err)
	}
	telemetry, ok := c.GCNativeCodeTelemetry()
	if !ok {
		t.Fatal("missing public telemetry")
	}
	report := map[string]any{"wasm_path": path, "wasm_sha256": hex.EncodeToString(wasmHash[:]), "native_sha256": hex.EncodeToString(codeHash.Sum(nil)), "compile_seconds": seconds, "fusions": total, "affected_functions": functions, "local_functions": len(stats.Funcs), "native_bytes": c.CodeSize(), "native_size": stats.NativeSize, "compile_resources": stats.Compile, "root_admission": roots, "gc_telemetry": telemetry, "functions": rows, "policy": "ordinary public Compile, CoreFeaturesV3, WithGCCodeTelemetry(true); no handwritten backend options"}
	encoded, err := json.MarshalIndent(report, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err = os.WriteFile(out, append(encoded, '\n'), 0644); err != nil {
		t.Fatal(err)
	}
	t.Logf("fusions=%d functions=%d native_bytes=%d compile_seconds=%.6f roots=%+v", total, functions, c.CodeSize(), seconds, roots)
	if total == 0 {
		t.Fatal("real compiler had zero admitted guard reuses")
	}
}
