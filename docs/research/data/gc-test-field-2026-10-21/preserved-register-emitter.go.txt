//go:build amd64

package amd64

import (
	"fmt"
	"math"

	"github.com/wago-org/wago/src/core/compiler/wasm"
	gc "github.com/wago-org/wago/src/core/runtime/gc/native"
)

type gcTestFieldPlan struct {
	ifPC, fieldPC, end int
	local              int
	offset, required   uint32
	scalar             directGCScalar
	ref, signed        bool
	cache              Reg
}

// Own only test; if; same-local final cast/get. The address is consumed at
// the start of the true arm, before ordinary compilation can emit any effect.
func (f *fn) planGCTestField(typeIndex uint32, reader *wasm.Reader) (p gcTestFieldPlan, ok bool) {
	if f.unreachable || f.inlineDepth != 0 || f.localBase != 0 || f.moduleEH || len(f.customInstructions) != 0 || f.intervalControl || f.vectorRegion.enabled || f.gcTestFieldSites >= 16 {
		return p, false
	}
	local, yes := gcLocalProvenance(f.s.back())
	if !yes {
		return p, false
	}
	st, yes := f.stagedStructType(typeIndex)
	if !yes || !st.Final {
		return p, false
	}
	cache := f.gcResolvedRegister()
	if cache == regNone {
		return p, false
	}
	rr := *reader
	window, err := rr.Bytes(min(64, rr.BytesLeft()))
	if err != nil {
		return p, false
	}
	r := wasm.ReaderFrom(window)
	byteIs := func(want byte) bool { b, e := r.Byte(); return e == nil && b == want }
	p.ifPC = reader.Offset()
	if !byteIs(0x04) {
		return p, false
	}
	b, yes := r.Peek()
	if !yes || !(b == 0x40 || isValByte(b) || b == 0x63 || b == 0x64) || !f.gcGuardBlockType(&r) {
		return p, false
	}
	if !byteIs(0x20) {
		return p, false
	}
	again, err := r.U32()
	if err != nil || int(again) != local || !byteIs(0xfb) {
		return p, false
	}
	cast, err := r.U32()
	if err != nil || cast != 22 && cast != 23 {
		return p, false
	}
	heap, exact, err := readRefHeapTypeImmediate(&r)
	if err != nil || exact || heap != int64(typeIndex) {
		return p, false
	}
	p.fieldPC = reader.Offset() + r.Offset()
	if !byteIs(0xfb) {
		return p, false
	}
	sub, err := r.U32()
	if err != nil || sub < 2 || sub > 4 {
		return p, false
	}
	typ, err := r.U32()
	if err != nil || typ != typeIndex {
		return p, false
	}
	field, err := r.U32()
	if err != nil {
		return p, false
	}
	fld, yes := f.stagedStructField(typeIndex, field)
	if !yes || (sub == 2) == fld.Storage().Packed() {
		return p, false
	}
	layout, final, yes := f.gcStructFieldLayout(typeIndex, field)
	if !yes || !final {
		return p, false
	}
	scalar, direct := directGCScalarStorage(fld.Storage())
	ref := layout.CollectorRef && layout.Size == 4
	if !ref && (!direct || scalar.typ.isFloat() || scalar.typ == mtV128 || layout.Size != uint32(scalar.size)) {
		return p, false
	}
	required := uint64(gc.PayloadOffset) + uint64(layout.Offset) + uint64(layout.Size)
	if required > math.MaxInt32 {
		return p, false
	}
	p.local, p.cache = local, cache
	p.offset, p.required = layout.Offset, uint32(required)
	p.scalar, p.ref, p.signed = scalar, ref, sub == 3
	p.end = reader.Offset() + r.Offset()
	return p, true
}

func (f *fn) tryGCTestField(typeIndex uint32, r *wasm.Reader) (bool, error) {
	p, ok := f.planGCTestField(typeIndex, r)
	if !ok {
		return false, nil
	}
	f.gcTestFieldSites++
	object := f.popValue()
	f.flush()
	ref := f.materialize(object)
	if ref != RAX {
		f.a.MovReg64(RAX, ref)
	}
	f.release(ref)
	f.a.MovImm32(RDX, int32(typeIndex))
	f.a.MovImm32(RCX, 0) // only non-null tests are admitted
	f.a.MovImm32(RSI, int32(p.required))
	if regallocCheckEnabled {
		f.checkCallClobber()
	}
	site := f.a.CallRel32()
	f.sc.gcTestFieldStubSites = append(f.sc.gcTestFieldStubSites, site)
	f.stats.call("gcnative")
	// The full test/conditional extent validation has returned EAX=boolean and
	// RDX=address (zero on false). No raw address crosses a call or safepoint.
	if f.regUser[p.cache] != nil || f.pinned.union(f.pinnedLocalMask).union(f.reserved).has(p.cache) {
		return true, fmt.Errorf("amd64: test/field register %d is no longer free", p.cache)
	}
	f.a.MovReg64(p.cache, RDX)
	f.gcResolved = gcResolvedObject{valid: true, local: p.local, typeIndex: typeIndex, requiredBytes: p.required, reg: p.cache}
	f.pinned = f.pinned.add(p.cache)
	defer f.invalidateGCResolvedObject()
	f.pushReg(RAX, mtI32)
	if err := r.JumpTo(p.ifPC + 1); err != nil {
		return true, err
	}
	oldPC := f.wasmPC
	defer func() { f.wasmPC = oldPC }()
	f.wasmPC = f.tracePCBase + uint32(p.ifPC)
	var previous profileOrigin
	if profileEnabled && f.stats != nil && f.stats.RecordSources {
		previous = f.enterProfileInstruction()
	}
	err := f.opBlock(r, 0x04)
	if profileEnabled && f.stats != nil && f.stats.RecordSources {
		f.switchProfileOrigin(previous)
	}
	if err != nil {
		return true, err
	}
	if !f.gcResolved.valid || f.gcResolved.reg != p.cache || !f.pinned.has(p.cache) {
		return true, fmt.Errorf("amd64: test/field lost address register %d at if entry", p.cache)
	}
	f.wasmPC = f.tracePCBase + uint32(p.fieldPC)
	if profileEnabled && f.stats != nil && f.stats.RecordSources {
		previous = f.enterProfileInstruction()
	}
	out := f.allocReg(maskOf(p.cache))
	if p.ref {
		f.a.Load32(out, p.cache, int32(gc.PayloadOffset+p.offset))
		f.setStackGCRoot(f.pushReg(out, mtI64), true)
	} else {
		f.a.LoadIdx(out, p.cache, RSP, int32(gc.PayloadOffset+p.offset), p.scalar.size, p.signed, p.scalar.typ == mtI64)
		f.pushReg(out, p.scalar.typ)
	}
	if profileEnabled && f.stats != nil && f.stats.RecordSources {
		f.switchProfileOrigin(previous)
	}
	if err := r.JumpTo(p.end); err != nil {
		return true, err
	}
	f.stats.peep("gc-test-field-fuse")
	return true, nil
}
