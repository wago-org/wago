package amd64

import "encoding/json"

// Archived compiler diagnostic sink only. No policy or code generation changes.
func corpusStats(ms *ModuleStats) string {
 out := struct {
  Functions int
  Report *GCLifetimeReport
  Lowered int
 }{Functions:len(ms.Funcs), Report:ms.GCLifetime}
 for _, f := range ms.Funcs { if f != nil { out.Lowered += f.Peephole["gc-lifetime-checked-scalar-replace"] } }
 b, _ := json.Marshal(out)
 return "CORPUS_STATS "+string(b)+"\n"
}
