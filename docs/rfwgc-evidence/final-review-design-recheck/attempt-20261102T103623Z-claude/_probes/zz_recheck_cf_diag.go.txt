package wago

// SCRATCH COUNTERFACTUAL ONLY (final-review-design-recheck diagnosis). Not a
// proposed repair: it substitutes the lease owner of the first domain in the
// callee's view for the callee's own pluginState identity, to separate an
// identity mismatch from incomplete root admission.
func recheckCFRootOwner(v gcInvocationDomainView) invocationID {
	if v.len() == 0 {
		return 0
	}
	d := v.at(0)
	d.invocationState.Lock()
	o := d.invocationOwner
	d.invocationState.Unlock()
	return o
}
