242: 	for _, plan := range p.plans[start:end] {
243: 		if plan.entry.space == spaceLarge {
244: 			continue
245: 		}
246: 		src := c.handles[plan.handle]
247: 		var dst []byte
248: 		if plan.entry.space == spaceNursery {
249: 			dst = c.nursery[plan.entry.off : plan.entry.off+plan.entry.size]
250: 		} else {
251: 			dst = c.throughput.bytes(plan.entry)
252: 		}
253: 		copy(dst, c.bytes(makeObjRef(plan.handle)))
254: 		if plan.entry.space == spaceOld {
255: 			binary.LittleEndian.PutUint32(dst[12:16], binary.LittleEndian.Uint32(dst[12:16])|FlagOld)
256: 		}
257: 		c.poisonYoungSource(src)
258: 	}
259: }

273: 	for range w.wake {

279: func (w *traceWorker) trace() {
280: 	p := w.pool
281: 	for {
282: 		for len(w.local) != 0 {
283: 			last := len(w.local) - 1
284: 			task := w.local[last]
285: 			w.local = w.local[:last]
286: 			w.tasks++
287: 			w.scan(task)
288: 			// Publication is batched. Private chains do no queue locking or waking.
289: 			if len(w.local) >= 32 && p.idle.Load() != 0 {
290: 				w.share()
291: 			}
292: 		}
293: 		p.mu.Lock()
294: 		p.active--
295: 		for len(p.queue) == 0 && p.active != 0 {
296: 			p.idle.Add(1)
297: 			p.ready.Wait()
298: 			p.idle.Add(-1)
299: 		}
300: 		if len(p.queue) == 0 {
301: 			p.ready.Broadcast()
302: 			p.mu.Unlock()
303: 			return
304: 		}
305: 		n := min(32, len(p.queue))
306: 		start := len(p.queue) - n
307: 		w.local = append(w.local, p.queue[start:]...)
308: 		p.queue = p.queue[:start]
309: 		p.active++
310: 		p.mu.Unlock()
311: 	}
312: }

314: func (w *traceWorker) share() {
315: 	p := w.pool
316: 	p.mu.Lock()
317: 	n := min(len(w.local)/2, parallelQueueLimit-len(p.queue))
318: 	if n != 0 {
319: 		start := len(w.local) - n
320: 		p.queue = append(p.queue, w.local[start:]...)
321: 		w.local = w.local[:start]
322: 		p.ready.Broadcast()
323: 	}
324: 	p.mu.Unlock()
325: }

327: func (w *traceWorker) mark(r Ref) {
328: 	if !r.IsObj() {
329: 		return
330: 	}
331: 	p := w.pool
332: 	c := p.collector
333: 	h := handleOf(r)
334: 	if h == 0 || int(h) >= len(c.handles) {
335: 		return
336: 	}
337: 	e := c.handles[h]
338: 	if e.space == spaceFree || p.nursery && !e.young() {
339: 		return
340: 	}
341: 	claim := &p.claims[h]
342: 	previous := atomic.LoadUint32(claim)
343: 	if previous == p.epoch || !atomic.CompareAndSwapUint32(claim, previous, p.epoch) {
344: 		return
345: 	}
346: 	c.mark[h] = true
347: 	w.local = append(w.local, traceTask{handle: h})
348: }

350: func (w *traceWorker) scan(task traceTask) {
351: 	c := w.pool.collector
352: 	r := makeObjRef(task.handle)
353: 	b := c.bytes(r)
354: 	if task.end != 0 {
355: 		for i := task.first; i < task.end; i++ {
356: 			w.mark(Ref(binary.LittleEndian.Uint32(b[PayloadOffset+i*4:])))
357: 		}
358: 		return
359: 	}
360: 	w.scanned++
361: 	if w.pool.capture {
362: 		w.live = append(w.live, task.handle)
363: 	}
364: 	typeID := binary.LittleEndian.Uint32(b)
365: 	if int(typeID) >= len(c.types) {
366: 		return
367: 	}
368: 	d := c.types[typeID]
369: 	if !d.HasRefs {
370: 		return
371: 	}
372: 	if d.Kind == KindStruct {
373: 		if nativeCollectorFastPaths && d.firstRef != 0 {
374: 			for index := uint32(d.firstRef); index != 0; {
375: 				field := d.Fields[index-1]
376: 				w.mark(Ref(binary.LittleEndian.Uint32(b[PayloadOffset+field.Offset:])))
377: 				index = uint32(field.nextRef)
378: 			}
379: 		} else {
380: 			for _, field := range d.Fields {
381: 				if isCollectorRefKind(field.Kind) {
382: 					w.mark(Ref(binary.LittleEndian.Uint32(b[PayloadOffset+field.Offset:])))
383: 				}
384: 			}
385: 		}
386: 	} else if d.ArrayElementsAreRefs() {
387: 		count := binary.LittleEndian.Uint32(b[8:])
388: 		// Reference payloads have a four-byte physical stride, for collector-managed refs.
389: 		for first := uint32(parallelArrayChunk); first < count; {
390: 			end := first + min(uint32(parallelArrayChunk), count-first)
391: 			w.local = append(w.local, traceTask{handle: task.handle, first: first, end: end})
392: 			first = end
393: 		}
394: 		if len(w.local) != 0 {
395: 			w.share()
396: 		}
397: 		for i := uint32(0); i < min(count, uint32(parallelArrayChunk)); i++ {
398: 			w.mark(Ref(binary.LittleEndian.Uint32(b[PayloadOffset+i*d.ElemSize:])))
399: 		}
400: 	}
401: }
