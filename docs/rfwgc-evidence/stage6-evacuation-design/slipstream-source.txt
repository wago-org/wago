452: fn ng_forward(h: u64, worker: u64, word: u64) -> u64 {
453:     if word == 0 || word & 1 != 0 { return word; }
454:     if !ng_valid(h, word) { ng_error(h, 2); return word; }
455:     let base = ng_get(h, N_HEAP); let mode = ng_get(h, N_MODE);
456:     let evacuate = mode == 2 && word >= NG_OLD && word < NG_LOS &&
457:         ng_get(h + N_BLOCKS + (word / NG_BLOCK) * 64, 24) != 0;
458:     if mode == 2 && !evacuate {
459:         if ng_set(h, N_MARKS, word) { ng_enqueue(h, worker, word); }
460:         return word;
461:     }
462:     if word >= NG_OLD && !evacuate { return word; }
463:     let to = ng_get(h, N_TO_BASE);
464:     if word >= to && word < to + ng_get(h, N_SURV_CAP) { return word; }
465:     var header = ss_ng_load32(base + word);
466:     loop {
467:         if header == NG_BUSY {
468:             if ng_get(h, N_ERROR) != 0 { return word; }
469:             ss_ng_pause(); header = ss_ng_load32(base + word); continue;
470:         }
471:         if header & NG_FORWARD != 0 { return header & ~NG_FORWARD; }
472:         let seen = ss_ng_cas32(base + word, header, NG_BUSY);
473:         if seen == header { break; }
474:         header = seen;
475:     }
476:     let id = header & 0xffffff; let age = (header >> 24) & 7;
477:     let aux = ss_ng_load32(base + word + 4); let size = ng_size(h, id, aux);
478:     let promote = evacuate || ng_get(h, N_PROMOTE_ALL) != 0 || age + 1 >= ng_get(h, N_AGE);
479:     let destination = if promote { ng_old_destination(h, worker, size) }
480:                       else { ng_survivor_destination(h, worker, size) };

483:     ss_ng_store32(base + destination, id | (if promote { 0 as u64 } else { age + 1 }) << 24);
484:     ss_ng_store32(base + destination + 4, aux);

1085: fn ng_trace_workers(h: u64, major: bool, total: u64, old_work: u64,
1086:                     local_major: bool, pressure: u64) -> u64 {
1087:     let available = ng_get(h, N_HELPERS) + 1;
1088:     // The production fault hooks name a helper and must never pass vacuously.
1089:     if ng_get(h, N_WORK_FAIL) != 0 || ng_get(h, N_ABORT_WORKER) != 0 { return available; }
1090:     if available == 1 || !major && total < NG_PARALLEL_BYTES { return 1; }
1091:     if major && pressure != 0 { return available; }
1092:     var work = min(total, ng_get(h, N_LAST_WORK));
1093:     if major { work = ng_saturating_add(work, old_work); }
1094:     var workers = min(available, 1 + work / NG_PAR_QUANTUM);
1095:     // Survivor history can lag, but an unbounded/reference-bearing major is
1096:     // never collapsed to the owner alone. This bounds the single-thread tail
1097:     // without forcing the whole admitted pool for intermediate work sizes.
1098:     if major && !local_major { workers = max(workers, min(available, 2)); }
1099:     return workers;
