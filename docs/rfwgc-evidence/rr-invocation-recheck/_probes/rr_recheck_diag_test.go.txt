package wago

// Diagnostic-only overlay: same table case as
// TestRRRecheckStorageOnlyReturnBehindDynamicReader, dumping all goroutine
// stacks while the storage-only call is blocked. Asserts nothing.

import (
	"runtime"
	"testing"
	"time"
)

func TestRRRecheckDiagStacks(t *testing.T) {
	f := rrStartDynamicSpin(t)
	s := rrInstantiate(t, rrCompile(t, rrModule(0x6f, false, false)), InstantiateOptions{store: f.rt.refStore})
	sdone := make(chan error, 1)
	go func() { _, e := s.Invoke("run"); sdone <- e }()
	select {
	case e := <-sdone:
		t.Logf("not blocked: %v", e)
	case <-time.After(2 * time.Second):
		buf := make([]byte, 1<<20)
		n := runtime.Stack(buf, true)
		t.Logf("BLOCKED; goroutines:\n%s", buf[:n])
		f.finish(t)
		t.Logf("after release: %v", <-sdone)
		return
	}
	f.finish(t)
}
