// Independent RR-1/RR-2 re-review adversaries (rr-invocation-recheck).
// Archived probe: copied into an immutable source export to execute. It uses
// only helpers that exist at 9713878ab, d2464100f and 9ad40bce4.

import (
	"context"
	"sync/atomic"
	"testing"
	"time"
	"unsafe"

	"github.com/wago-org/wago/tests/support/wasmtest"
)

// rrModule: export "run" () -> i32. table is 0 (none), 0x6f (externref, storage
// only) or 0x70 (funcref, true-dynamic). spin waits on shared memory [0] after
// storing 1 at [4]; otherwise run returns 7 at once. importA imports u.a.
func rrModule(table byte, spin, importA bool) []byte {
	var imps [][]byte
	if spin {
		imps = append(imps, append(append(wasmtest.Name("env"), wasmtest.Name("memory")...), 2, 3, 1, 1))
	}
	if importA {
		imps = append(imps, append(append(wasmtest.Name("u"), wasmtest.Name("a")...), 0, 1))
	}
	sections := [][]byte{wasmtest.Section(1, wasmtest.Vec([]byte{0x60, 0, 1, 0x7f}, []byte{0x60, 0, 0}))}
	if len(imps) != 0 {
		sections = append(sections, wasmtest.Section(2, wasmtest.Vec(imps...)))
	}
	sections = append(sections, wasmtest.Section(3, wasmtest.Vec([]byte{0})))
	if table != 0 {
		sections = append(sections, wasmtest.Section(4, wasmtest.Vec([]byte{table, 0, 1})))
	}
	body := []byte{0x41, 7, 0x0b}
	if spin {
		body = []byte{0x41, 4, 0x41, 1, 0xfe, 0x17, 2, 0,
			0x03, 0x40, 0x41, 0, 0xfe, 0x10, 2, 0, 0x45, 0x0d, 0, 0x0b, 0x41, 7, 0x0b}
	}
	run := byte(0)
	if importA {
		run = 1
	}
	sections = append(sections, wasmtest.Section(7, wasmtest.Vec(wasmtest.ExportEntry("run", 0, run))), wasmtest.Section(10, wasmtest.Vec(wasmtest.Code(body))))
	return wasmtest.Module(sections...)
}

func rrCompile(t *testing.T, raw []byte) *Compiled {
	t.Helper()
	c, err := Compile(NewRuntimeConfig().WithCoreFeatures(CoreFeaturesV3|CoreFeatureThreads).WithBoundsChecks(BoundsChecksExplicit), raw)
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = c.Close() })
	return c
}

func rrInstantiate(t *testing.T, c *Compiled, opts InstantiateOptions) *Instance {
	t.Helper()
	in, err := instantiateCore(c, opts)
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { in.Close() })
	return in
}

type rrFixture struct {
	rt      *Runtime
	u1      *Instance
	dyn     *Instance
	flag    *uint32
	entered *uint32
	done    chan error
}

// A Runtime with one GC tenant, and a true-dynamic (funcref table) guest call
// spinning while it holds its topology reader across native execution.
func rrStartDynamicSpin(t *testing.T) *rrFixture {
	t.Helper()
	memory, err := NewSharedMemory(1, 1)
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { memory.Close() })
	f := &rrFixture{rt: NewRuntime(), done: make(chan error, 1)}
	uc, err := Compile(NewRuntimeConfig().WithCoreFeatures(CoreFeaturesV3), wasmtest.Module(
		wasmtest.Section(1, wasmtest.Vec([]byte{0x5f, 1, 0x7f, 1}, []byte{0x60, 0, 0})),
		wasmtest.Section(3, wasmtest.Vec([]byte{1})),
		wasmtest.Section(7, wasmtest.Vec(wasmtest.ExportEntry("a", 0, 0))),
		wasmtest.Section(10, wasmtest.Vec(wasmtest.Code([]byte{0x41, 1, 0xfb, 0, 0, 0x1a, 0x0b})))))
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = uc.Close() })
	f.u1 = rrInstantiate(t, uc, InstantiateOptions{store: f.rt.refStore, GC: GCConfig{NurseryBytes: 8192}})
	f.dyn = rrInstantiate(t, rrCompile(t, rrModule(0x70, true, false)), InstantiateOptions{store: f.rt.refStore, Imports: testImports("env.memory", memory)})
	mem := memory.UnsafeBytes()
	f.flag = (*uint32)(unsafe.Pointer(&mem[0]))
	f.entered = (*uint32)(unsafe.Pointer(&mem[4]))
	go func() { _, e := f.dyn.Invoke("run"); f.done <- e }()
	deadline := time.Now().Add(5 * time.Second)
	for atomic.LoadUint32(f.entered) == 0 {
		if time.Now().After(deadline) {
			atomic.StoreUint32(f.flag, 1)
			t.Fatal("dynamic spin never entered")
		}
		time.Sleep(time.Millisecond)
	}
	return f
}

func (f *rrFixture) finish(t *testing.T) {
	t.Helper()
	atomic.StoreUint32(f.flag, 1)
	if err := <-f.done; err != nil {
		t.Fatalf("dynamic spin: %v", err)
	}
}

func rrFlags(in *Instance) (dynamic, conversionOnly bool) {
	dynamic = in.executionFlags.Load()&executionFlagDynamicGCDomain != 0
	in.refStore.mu.Lock()
	defer in.refStore.mu.Unlock()
	if entry := in.refStore.instances[in]; entry != nil {
		conversionOnly = rrConversionOnly(entry)
	}
	return
}

// A storage-only (or scalar) call that never waits for a domain must return
// while an unrelated true-dynamic call runs. A second storage-only admission
// must not stall behind the first one's release.
func TestRRRecheckStorageOnlyReturnBehindDynamicReader(t *testing.T) {
	for _, kind := range []string{"none", "table"} {
		t.Run(kind, func(t *testing.T) {
			f := rrStartDynamicSpin(t)
			table := byte(0)
			if kind == "table" {
				table = 0x6f
			}
			sc := rrCompile(t, rrModule(table, false, false))
			s := rrInstantiate(t, sc, InstantiateOptions{store: f.rt.refStore})
			s2 := rrInstantiate(t, sc, InstantiateOptions{store: f.rt.refStore})
			sdone := make(chan error, 1)
			go func() {
				out, e := s.Invoke("run")
				if e == nil && (len(out) != 1 || out[0] != 7) {
					t.Errorf("run result %v", out)
				}
				sdone <- e
			}()
			blocked := false
			var serr error
			select {
			case serr = <-sdone:
			case <-time.After(2 * time.Second):
				blocked = true
			}
			ctx, cancel := context.WithTimeout(context.Background(), time.Second)
			_, s2err := s2.InvokeContext(ctx, "run")
			cancel()
			f.finish(t)
			if blocked {
				serr = <-sdone
			}
			if serr != nil {
				t.Fatalf("storage call: %v", serr)
			}
			dd, dc := rrFlags(f.dyn)
			sd, sco := rrFlags(s)
			t.Logf("kind=%s dynSpin(dynamic=%t conversionOnly=%t) storage(dynamic=%t conversionOnly=%t) storageReturnBlocked=%t secondAdmissionErr=%v", kind, dd, dc, sd, sco, blocked, s2err)
			if blocked {
				t.Errorf("storage-only call could not return while an unrelated true-dynamic call ran")
			}
			if s2err != nil {
				t.Errorf("second storage-only call stalled: %v", s2err)
			}
		})
	}
}

// A canceled storage-only admission that waits on a domain gate must return
// promptly with its context error.
func TestRRRecheckCanceledAdmissionBehindDynamicReader(t *testing.T) {
	f := rrStartDynamicSpin(t)
	a, err := f.u1.ExportedFunc("a")
	if err != nil {
		t.Fatal(err)
	}
	s := rrInstantiate(t, rrCompile(t, rrModule(0x6f, false, true)), InstantiateOptions{store: f.rt.refStore, Imports: testImports("u.a", a)})
	ctx, cancel := context.WithTimeout(context.Background(), 300*time.Millisecond)
	defer cancel()
	sdone := make(chan error, 1)
	go func() { _, e := s.InvokeContext(ctx, "run"); sdone <- e }()
	blocked := false
	var serr error
	select {
	case serr = <-sdone:
	case <-time.After(2300 * time.Millisecond):
		blocked = true
	}
	f.finish(t)
	if blocked {
		serr = <-sdone
	}
	sd, sco := rrFlags(s)
	t.Logf("storage(dynamic=%t conversionOnly=%t) canceledReturnBlocked=%t err=%v", sd, sco, blocked, serr)
	if blocked {
		t.Errorf("canceled storage-only admission did not return within 2s of its deadline")
	}
	if serr == nil {
		t.Errorf("admission behind a held domain gate unexpectedly succeeded")
	}
}
