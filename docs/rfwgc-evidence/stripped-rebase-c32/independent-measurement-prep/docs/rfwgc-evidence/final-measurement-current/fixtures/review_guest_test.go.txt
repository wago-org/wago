// No GC types or conversion opcodes. A non-null ordinary host token exercises
// helper routing without conversion authority. Bulk count is exactly 256.
func fmExternModule(op string) []byte {
 i32:=func(n int) []byte {return append([]byte{0x41},wasmtest.SLEB32(int32(n))...)}
 body:=[]byte{}
 switch op {
 case "global-set":for i:=0;i<256;i++ {body=append(body,0x20,0,0x24,0)};body=append(body,0x23,0,0xd1)
 case "table-set":for i:=0;i<256;i++ {body=append(body,i32(i)...);body=append(body,0x20,0,0x26,0)};body=append(body,i32(255)...);body=append(body,0x25,0,0xd1)
 case "fill","copy","init":
  body=append(body,i32(0)...);body=append(body,0x20,0);body=append(body,i32(256)...);body=append(body,0xfc,17,0)
  if op=="copy" {body=append(body,i32(256)...);body=append(body,i32(0)...);body=append(body,i32(256)...);body=append(body,0xfc,14,0,0)}
  if op=="init" {body=append(body,i32(0)...);body=append(body,i32(0)...);body=append(body,i32(256)...);body=append(body,0xfc,12,0,0)}
  last:=255;if op=="copy" {last=511};body=append(body,i32(last)...);body=append(body,0x25,0,0xd1)
 case "grow":body=append(body,0x20,0);body=append(body,i32(256)...);body=append(body,0xfc,15,0)
 default:panic(op)
 }
 body=append(body,0x0b)
 exprs:=make([][]byte,256);for i:=range exprs {exprs[i]=[]byte{0xd0,0x6f,0x0b}}
 elem:=append([]byte{5,0x6f},wasmtest.Vec(exprs...)...)
 return wasmtest.Module(
  wasmtest.Section(1,wasmtest.Vec([]byte{0x60,1,0x6f,1,0x7f})),wasmtest.Section(3,wasmtest.Vec([]byte{0})),
  wasmtest.Section(4,wasmtest.Vec(append([]byte{0x6f,0},wasmtest.ULEB(512)...))),
  wasmtest.Section(6,wasmtest.Vec([]byte{0x6f,1,0xd0,0x6f,0x0b})),wasmtest.Section(7,wasmtest.Vec(wasmtest.ExportEntry("run",0,0))),
  wasmtest.Section(9,wasmtest.Vec(elem)),wasmtest.Section(10,wasmtest.Vec(wasmtest.Code(body))))
}
func fmExternCycle(t testing.TB,op string) {
 c,err:=Compile(NewRuntimeConfig().WithCoreFeatures(CoreFeaturesV3),fmExternModule(op));if err!=nil {t.Fatal(err)};defer c.Close()
 in,err:=Instantiate(c,InstantiateOptions{});if err!=nil {t.Fatal(err)};defer in.Close()
 if in.gc!=nil {t.Fatal("non-GC workload installed a collector")}
 ref,err:=in.NewExternRef("ordinary-host-token");if err!=nil {t.Fatal(err)};defer in.ReleaseExternRef(ref)
 got,err:=in.Invoke("run",ref.token);want:=uint64(0);if op=="init" {want=1};if op=="grow" {want=512}
 if err!=nil||len(got)!=1||got[0]!=want {t.Fatalf("%s: %v %v want %d",op,got,err,want)}
}
func TestFinalMeasurementPlainExtern(t *testing.T) {for _,op:=range []string{"global-set","table-set","fill","copy","init","grow"} {t.Run(op,func(t *testing.T){fmExternCycle(t,op)})}}
// Full fresh-instance cycles, including compile, host-token creation and close.
// Do not label these isolated store latencies.
func BenchmarkFinalMeasurementPlainExtern(b *testing.B) {for _,op:=range []string{"global-set","table-set","fill","copy","init","grow"} {b.Run(op,func(b *testing.B){for i:=0;i<b.N;i++ {fmExternCycle(b,op)}})}}
