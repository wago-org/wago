        0 => {
            if !store_function_subtype(store, e.type_id, type_ids[t.index as i64]) { return 129; }
            m.funcs.push(t.index); m.imported_funcs += 1; refs.push(e.address);
        },
        1 => {
            let expected = runtime_value_type(m, t.value_type); let address = if t.wide { 0x7e as u8 } else { 0x7f as u8 };
            if expected != e.value_type || e.address_type != address { return 129; }
            let descriptor = e.address;
            if ss_load64(descriptor + 8) < t.minimum ||
               t.has_maximum && (ss_load64(descriptor + 24) & 2 == 0 || ss_load64(descriptor + 16) > t.maximum) { return 129; }
            m.tables.push(t.value_type); m.table_types.push(address); table_refs.push(descriptor);
            m.imported_tables += 1;
            if rt.native_heap != 0 {
                let kind = root_value_kind(m, t.value_type);
                var native_region = ng_region_key(rt.native_heap, descriptor);
                if native_region == 0 && kind != 0 {
                    native_region = gc2_region_install(rt, ss_load64(descriptor),
                        ss_load64(descriptor + 8), 8, kind, descriptor);
                }
                require(r, kind == 0 || native_region != 0, "imported table root registration failed");
                table_regions.push(native_region);
            }
        },
        2 => {
            let descriptor = e.address;
            let flags = (if t.wide { 1 as u64 } else { 0 as u64 }) | (if t.shared { 2 as u64 } else { 0 as u64 });
            if ss_load64(descriptor + 24) & 3 != flags || (ss_load64(descriptor + 8) >> 16) < t.minimum ||
               t.has_maximum && (ss_load64(descriptor + 16) >> 63 == 0 ||
                   (ss_load64(descriptor + 16) & 9223372036854775807) > t.maximum) { return 129; }
            memory_refs.push(descriptor); m.memory_types.push(if t.wide { 0x7e } else { 0x7f }); m.memories += 1;
        },
        3 => {
            let expected = runtime_value_type(m, t.value_type);
            if t.mutable != e.mutable || (if t.mutable { expected != e.value_type }
                else { !store_value_matches(store, e.value_type, expected) }) { return 129; }
            global_refs.push(e.address);
            if rt.native_heap != 0 {
                let kind = root_value_kind(m, t.value_type);
                var native_region = ng_region_key(rt.native_heap, e.address);
                if native_region == 0 && kind != 0 {
                    native_region = gc2_region_install(rt, e.address, 1, 16, kind, e.address);
                }
                require(r, kind == 0 || native_region != 0, "imported global root registration failed");
                global_regions.push(native_region);
            }
            m.globals.push(Global { t.value_type, t.mutable }); m.imported_globals += 1;
        },
        4 => {
            if type_ids[t.index as i64] != e.type_id { return 129; }
            m.tags.push(Tag { t.index }); m.imported_tags += 1; tag_refs.push(e.address);
        },
        _ => { return 129; }
    }
    return 0;
}
fn finalize_host_imports(r: Reader&, store: Store&, m: Module&, rt: Runtime&,
                         type_ids: u32[>..]&, refs: u64[>..]&, host_imports: u32[>..]&) {
    require(r, host_imports.len % 2 == 0, "invalid pending host imports");
    let count = host_imports.len / 2;
    let base = store_region(store, r, count as u64 * 48);
    rt.host_function_base = base; rt.host_function_count = count as u64;
    for i in count {
        let ref_index = host_imports[i * 2];
        let operation = host_imports[i * 2 + 1];
        let type_index = m.funcs[ref_index as i64];
        let sig = signature(r, m, type_index);
        let reference = base + i as u64 * 48;
        ss_store64(reference + 8, ss_runtime_pointer(rt));
        ss_store64(reference + 16, type_ids[type_index as i64]);
        var encoded_operation = operation as u64;
        if operation < 261 && sig.results.len != 0 {
            let result = m.value_types[sig.results.pos];
            if is_ref_type(result) && heap_defined(ref_heap(result)) {
                encoded_operation |= ((heap_type_index(ref_heap(result)) as u64 + 1) << 32);
            }
        }
        ss_store64(reference + 24, encoded_operation);
        ss_store64(reference + 32, sig.params.len); ss_store64(reference + 40, sig.results.len);
        refs[ref_index as i64] = reference;
    }
}
fn load_runtime_module<Register>(r: Reader&, m: Module&, rt: Runtime&, bodies: FunctionBody[>..]&,
                       data: DataSegment[>..]&, elements: ElementSegment[>..]&,
                       export_names: u8[>..]&, runtime_exports: RuntimeExport[>..]&, target: const u8[:],
                       store: Store&, refs: u64[>..]&, global_refs: u64[>..]&,
                       table_refs: u64[>..]&, global_regions: u64[>..]&, table_regions: u64[>..]&,
                       element_regions: u64[>..]&, type_ids: u32[>..]&, initializers: Initializer[>..]&) -> i64, i64, i64 {
    skip(r, 8); var selected: i64 = -1; var start: i64 = -1;
    rt.fd = r.fd; var memory_refs: u64[>..] = []; var own_memories: u64[>..] = [];
    var tag_refs: u64[>..] = [];
    var registered = false;
    while position(r) < r.size {
        let id = byte(r); let size = uleb(r) as i64;
        require(r, size <= r.size - position(r), "truncated section"); set_end(r, position(r) + size);
        // Type sections precede every section that can evaluate a managed
        // constant or publish managed storage.  Modules without one still
        // need their collector and initially empty type map before decoding
        // their first semantic section.
        if id != 0 && id != 1 && !registered {
            Register(r, store, m, rt); registered = true;
        }
        match id {
            0 => { skip_name(r); skip(r, r.end - position(r)); },
            1 => {
                read_type_section(r, m);
                intern_types(r, store, m, type_ids);
                Register(r, store, m, rt);
                registered = true;
            },
            2 => {
                // Keep only compact indexes while the streamed section is open,
