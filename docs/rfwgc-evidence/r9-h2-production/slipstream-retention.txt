fn recycle_store_file(store: Store&, index: i64) {
    ss_input_close(store.files[index]); store.files[index] = -store.free_file - 2; store.free_file = index;
}
inline fn failed_runtime_reference(rt: Runtime&, value: u64) -> bool {
    return value != 0 &&
        (rt.own_function_count != 0 && value >= rt.own_functions &&
         value < rt.own_functions + rt.own_function_count * 24 && (value - rt.own_functions) % 24 == 0 ||
         rt.host_function_count != 0 && value >= rt.host_function_base &&
         value < rt.host_function_base + rt.host_function_count * 48 && (value - rt.host_function_base) % 48 == 0);
}
fn failed_runtime_words_reference(rt: Runtime&, base: u64, count: u64) -> bool {
    for i in count as i64 { if failed_runtime_reference(rt, ss_load64(base + (i as u64) * 8)) { return true; } }
    return false;
}
fn failed_runtime_reachable(store: Store&, failed_index: i64) -> bool {
    let failed: Runtime& = store.runtimes[failed_index];
    var nested_native_edge = false;
    for runtime_index in store.runtimes.len {
        if runtime_index == failed_index { continue; }
        let rt: Runtime& = store.runtimes[runtime_index];
        if rt.function_refs != 0 && failed_runtime_words_reference(failed, rt.function_refs, rt.function_ref_count) { return true; }
        for i in rt.gc_global_count {
            if failed_runtime_reference(failed, ss_load64(ss_load64(rt.global_refs + i * 8))) { return true; }
        }
        for i in rt.gc_table_count {
            let d = ss_load64(rt.table_refs + i * 8);
            if failed_runtime_words_reference(failed, ss_load64(d), ss_load64(d + 8)) { return true; }
        }
        for i in rt.gc_element_count {
            let d = rt.elements + i * 24;
            if failed_runtime_words_reference(failed, ss_load64(d), ss_load64(d + 8)) { return true; }
        }
        if rt.gc_global_regions != 0 || rt.gc_table_regions != 0 || rt.gc_element_regions != 0 {
            nested_native_edge = true;
        }
    }
    if failed.gc_runtime != 0 && failed.native_heap == 0 && failed.gc_roots_retained != 0 {
        return ss_gc_failed_runtime_reachable_pointer(failed.gc_runtime, failed);
    }
    // ABI2 requires its exact collector trace; until that trace reports no
    // external edge, retaining is the only sound result.
    // ABI2 function references can be nested collector fields below private
    // provider storage, where no direct slot contains the native pointer.
    // Until ABI2 exposes an exact failure trace, any earlier live runtime can
    // carry such an edge and makes rollback unsafe.
    if store.gc_runtime != 0 && ss_load64(store.gc_runtime) == 2 && nested_native_edge { return true; }
    return failed.native_heap != 0 &&
           (failed.gc_global_count != 0 || failed.gc_table_count != 0 || failed.gc_element_count != 0);
}
