585: fn ng_ack_phase(h: u64, worker: u64, ticket: u64) -> bool {
586:     let at = h + 1024 + worker * 8; var seen = ss_ng_load64(at);
587:     loop {
588:         if seen == ticket { return true; }
589:         if seen > ticket { ng_fail_closed(h, 13); return false; }
590:         let old = ss_ng_cas(at, seen, ticket);
591:         if old == seen { return true; }
592:         seen = old;
593:     }
594: }
595: // N_DONE is (ticket << 8) | completed_helpers. The completion CAS is the last
596: // shared phase operation. A following futex wake is notification only: an old
597: // delayed wake is harmless because the owner always rereads the full word.
598: fn ng_complete(h: u64, worker: u64, ticket: u64, workers: u64) -> bool {
599:     if !ng_ack_phase(h, worker, ticket) { ss_ng_wake(h + N_DONE); return false; }
600:     let bit = (1 as u64) << worker; let target = ng_done_target(workers);
601:     loop {
602:         let done = ng_get(h, N_DONE);
603:         if done >> 8 != ticket { ng_fail_closed(h, 13); ss_ng_wake(h + N_DONE); return false; }
604:         if done & bit != 0 { return true; }
605:         let next = done | bit; let seen = ss_ng_cas(h + N_DONE, done, next);
606:         if seen != done { continue; }
607:         if next & 255 == target { ss_ng_wake(h + N_DONE); }
608:         return true;
609:     }
610: }

979:     ng_put(h, N_RUNS, 0); ng_put(h, N_RUN_CLAIM, 0);
980:     if bytes == 0 { return true; }

982:     if bytes > 0xffffffffffffffff - workers * NG_BLOCK {

993:       if available == 0 { continue; }
994:       for bit in 64 {
995:         if available & ((1 as u64) << bit) == 0 { continue; }
996:         let block_index = bitmap_word * 64 + bit as u64;
997:         let b = h + N_BLOCKS + block_index * 64; var line: u64 = 0;
998:         if ng_get(b, 24) != 0 { continue; }
999:         while line < 128 {
1000:             if ss_ng_load64(b + 8 + (line / 64) * 8) & ((1 as u64) << (line & 63)) != 0 { line += 1; continue; }
1001:             let first = line;
1002:             while line < 128 && ss_ng_load64(b + 8 + (line / 64) * 8) & ((1 as u64) << (line & 63)) == 0 { line += 1; }
1003:             let size = (line - first) * NG_LINE;
1004:             if size <= maximum { continue; }
1005:             if ng_get(b, 0) == 0 {
1006:                 if ng_fault(h) || ss_ng_protect(ng_get(h, N_HEAP) + block_index * NG_BLOCK, NG_BLOCK, 3) != 0 {
1007:                     ng_put(h, N_ADMIT_CREDIT, credit);
1008:                     if required { ng_refuse(h, NG_REFUSAL_PROTECT); }
1009:                     return false;
1010:                 }
1011:                 ng_put(b, 0, 1); high = max(high, (block_index + 1) * NG_BLOCK); ng_put(h, N_OLD_HIGH, high);
1012:             }
1013:             let n = ng_get(h, N_RUNS);
1014:             if n >= 65536 {
1015:                 ng_put(h, N_ADMIT_CREDIT, credit);
1016:                 if required { ng_refuse(h, NG_REFUSAL_RUN_TABLE); }
1017:                 return false;
1018:             }
1019:             ng_put(h, N_RUN_DATA + n * 16, block_index * NG_BLOCK + first * NG_LINE);
1020:             ng_put(h, N_RUN_DATA + n * 16 + 8, block_index * NG_BLOCK + line * NG_LINE);
1021:             ng_put(h, N_RUNS, n + 1); credit += size - maximum; ng_put(h, N_ADMIT_CREDIT, credit);
1022:             if credit >= need { return true; }
1023:         }
1024:       }
1025:     }
