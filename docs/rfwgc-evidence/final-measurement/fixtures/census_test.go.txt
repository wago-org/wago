package gc
import ("testing";"reflect";"encoding/json")
// Diagnostic-only snapshots, never called by timing benchmark bodies. Backing
// len/cap is an endpoint census, NOT RSS, peak occupancy or fragmentation.
func fmDetailedCensus(t *testing.T,c *Collector,label string) {
 row:=map[string]any{"label":label,"stats":c.Stats(),"nursery_bump":c.nurseryBump,"nursery_backing_len":len(c.nursery),"nursery_backing_cap":cap(c.nursery),"old_los_backing_len":len(c.throughput.mem),"old_los_backing_cap":cap(c.throughput.mem),"old_los_limit":c.throughput.limit,"old_los_bump":c.throughput.bump,"free_bytes":c.throughput.freeBytes,"pending_bytes":c.throughput.pendingBytes,"free_runs":c.throughput.spanCount}
 occupancy:=map[uint8]uint64{};handles:=map[uint8]uint64{}
 for _,e:=range c.handles {if e.space!=spaceFree {occupancy[uint8(e.space)]+=uint64(e.size);handles[uint8(e.space)]++}}
 row["live_payload_bytes_by_space"]=occupancy;row["live_handles_by_space"]=handles;row["space_key"]="1 nursery; 2 old; 3 LOS; 4 Tiny"
 value:=reflect.ValueOf(c).Elem();regions:=value.FieldByName("edenRegions")
 if regions.IsValid(){used:=[]uint64{};for i:=0;i<regions.Len();i++{used=append(used,regions.Index(i).FieldByName("bump").Uint())};row["eden_used_extents"]=used}
 mature:=value.FieldByName("throughput").FieldByName("mature")
 if mature.IsValid()&&!mature.IsNil(){regions:=mature.Elem().FieldByName("regions");sums:=map[string]uint64{};for i:=0;i<regions.Len();i++ {for _,name:=range []string{"live","large","reserved","reservedLarge"}{sums[name]+=regions.Index(i).FieldByName(name).Uint()}};row["mature_descriptor_count"]=regions.Len();row["mature_descriptor_cap"]=regions.Cap();row["mature_physical_bytes"]=sums}
 for _,name:=range []string{"ParallelActivity","CollectionDiagnostics"} {method:=reflect.ValueOf(c).MethodByName(name);if method.IsValid(){row[name]=method.Call(nil)[0].Interface()}}
 raw,err:=json.Marshal(row);if err!=nil {t.Fatal(err)};t.Log("FM_CENSUS",string(raw))
}
