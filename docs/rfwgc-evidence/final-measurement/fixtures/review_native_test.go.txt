// 8192 * 8192 payload bytes retained in old space (64 MiB), then 8192
// ephemeral scalar allocations against 64 KiB Eden. Construction is explicit
// promotion; timed-cycle scope includes setup/validation/cleanup, not pure pause.
func fmOldFill(t testing.TB) {
 shift,_:=strconv.Atoi(os.Getenv("FM_SHIFT"));workers,_:=strconv.Atoi(os.Getenv("FM_WORKERS"));if workers==0 {workers=1}
 cfg:=Config{NurseryBytes:64<<10,ThroughputHeapBytes:160<<20}
 for k,n:=range map[string]int{"NurseryRegionShift":shift,"Workers":workers} {f:=reflect.ValueOf(&cfg).Elem().FieldByName(k);if f.IsValid(){if f.Kind()==reflect.Int {f.SetInt(int64(n))}else{f.SetUint(uint64(n))}}else if n>1||k=="NurseryRegionShift"&&n!=0 {t.Fatal("unsupported",k,n)}}
 arr,err:=NewArrayDesc(0,StorageI8);if err!=nil {t.Fatal(err)};scalar,err:=NewStructDesc(1,[]StorageKind{StorageI64});if err!=nil {t.Fatal(err)}
 c,err:=NewCollector(cfg,[]TypeDesc{arr,scalar});if err!=nil {t.Fatal(err)};defer c.Close()
 roots:=make(RefSliceRoots,0,8192)
 for i:=0;i<8192;i++ {r,err:=c.NewArrayDefaultWithRoots(0,8192,roots);if err!=nil {t.Fatal(err)};if err=c.promoteHandle(handleOf(r));err!=nil {t.Fatal(err)};roots=append(roots,r)}
 if err=c.CollectFull(roots);err!=nil {t.Fatal(err)}
 edges:=&fmReviewEdges{};if os.Getenv("FM_OBJECT_EDGES")=="1" {method:=reflect.ValueOf(c).MethodByName("SetObjectEdgeSource");if !method.IsValid(){t.Skip("object-edge source unavailable at pin")};method.Call([]reflect.Value{reflect.ValueOf(edges)})}
 before:=c.Stats()
 methodActivity:=reflect.ValueOf(c).MethodByName("ParallelActivity");var beforeActivity reflect.Value
 if methodActivity.IsValid(){beforeActivity=methodActivity.Call(nil)[0]}
 if test,ok:=t.(*testing.T);ok {fmDetailedCensus(test,c,"old-established-before-fill")}
 for i:=0;i<8192;i++ {if _,err=c.NewStructDefaultWithRoots(1,roots);err!=nil {t.Fatal(err)}}
 if test,ok:=t.(*testing.T);ok {fmDetailedCensus(test,c,"after-fill-before-explicit-full")}
 after:=c.Stats();if after.Allocations-before.Allocations!=8192 {t.Fatal("allocation census",before,after)}
 if err=c.CollectFull(roots);err!=nil {t.Fatal(err)};if c.Stats().LiveObjects!=8192 {t.Fatal("live census",c.Stats())};if err=c.Verify(roots);err!=nil {t.Fatal(err)}
 if os.Getenv("FM_OBJECT_EDGES")=="1"&&methodActivity.IsValid(){afterActivity:=methodActivity.Call(nil)[0];for i:=1;i<8;i++ {if afterActivity.FieldByName("Trace").Index(i).Uint()!=beforeActivity.FieldByName("Trace").Index(i).Uint(){t.Fatal("object-edge trace unexpectedly used parallel participants")}}}
 if test,ok:=t.(*testing.T);ok {
  fmDetailedCensus(test,c,"after-explicit-full")
  if beforeActivity.IsValid(){raw,_:=json.Marshal(beforeActivity.Interface());test.Log("before_joined_phase_participants",string(raw))}
  raw,_:=json.Marshal(map[string]any{"before":before,"after_fill":after,"after_full":c.Stats(),"object_edge_calls":edges.calls,"requested_workers":workers});test.Log(string(raw))
  // Reflection retains baseline build compatibility; missing counters stay absent.
  method:=reflect.ValueOf(c).MethodByName("ParallelActivity");if method.IsValid(){raw,_=json.Marshal(method.Call(nil)[0].Interface());test.Log("joined_phase_participants",string(raw))}
 }
 if os.Getenv("FM_OBJECT_EDGES")=="1"&&edges.calls==0 {t.Fatal("object edge source unused")}
 if err=c.CollectFull(nil);err!=nil {t.Fatal(err)};if c.Stats().LiveObjects!=0 {t.Fatal("cleanup")}
}
func TestFinalMeasurementLargeOldNurseryFill(t *testing.T){fmOldFill(t)}
func BenchmarkFinalMeasurementLargeOldNurseryFill(b *testing.B){for i:=0;i<b.N;i++ {fmOldFill(b)}}
