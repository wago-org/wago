package gc
import ("encoding/json"; "os"; "strconv"; "testing")
func fmEvacHeap(t testing.TB, workers, count int, shift uint8, enabled bool) (*Collector, RefSliceRoots) {
	t.Helper()
	c, err := NewCollector(Config{Workers: workers, NurseryBytes: 1 << 20, NurseryRegionShift: shift, MatureEvacuation: enabled, ThroughputHeapBytes: 32 << 20, ThroughputPageBytes: 4096, PoisonFreed: true, VerifyAfterCollect: true}, testTypes(t))
 if err != nil { t.Fatal(err) }
	refs := make(RefSliceRoots, 0, count)
	var gaps []handleEntry
	for i := 0; i < count; i++ {
		r, err := c.NewStructDefault(0)
		if err != nil {
			t.Fatal(err)
		}
		if err = c.promoteHandle(handleOf(r)); err != nil {
			t.Fatal(err)
		}
		refs = append(refs, r)
		if (i+1)%128 == 0 {
			gap, err := c.throughput.alloc(32768-4096, spaceOld)
			if err != nil {
				t.Fatal(err)
			}
			gaps = append(gaps, gap)
		}
	}
	for _, e := range gaps {
		if err := c.throughput.free(e); err != nil {
			t.Fatal(err)
		}
	}
	if err := c.Verify(refs); err != nil {
		t.Fatal(err)
	}
	return c, refs
}


func fmEvac(t testing.TB) {
 workers,_:=strconv.Atoi(os.Getenv("FM_WORKERS")); if workers==0 {workers=1}
 enabled:=os.Getenv("FM_EVAC")=="On"
 shift,_:=strconv.Atoi(os.Getenv("FM_SHIFT")); if shift==0 {shift=12}
 c,roots:=fmEvacHeap(t,workers,8448,uint8(shift),enabled); defer c.Close()
 c.throughput.mature.pressure=true
 if err:=c.CollectFull(roots);err!=nil {t.Fatal(err)}
 d:=c.CollectionDiagnostics()
 if enabled && (d.Evacuation!="moved" || d.MovedBytes!=uint64(len(roots))*32) {t.Fatal("no actual movement",d)}
 if !enabled && d.MovedBytes!=0 {t.Fatal("unexpected movement",d)}
 if c.Stats().LiveObjects!=uint32(len(roots)) {t.Fatal("census",c.Stats())}
 if err:=c.Verify(roots);err!=nil {t.Fatal(err)}
 if test,ok:=t.(*testing.T);ok {raw,_:=json.Marshal(d);test.Log(string(raw))}
 if err:=c.CollectFull(nil);err!=nil {t.Fatal(err)}
 if c.Stats().LiveObjects!=0 {t.Fatal("cleanup")}
}
func TestFinalMeasurementEvacuation(t *testing.T) {fmEvac(t)}
func BenchmarkFinalMeasurementEvacuation(b *testing.B) {b.ReportAllocs();for i:=0;i<b.N;i++ {fmEvac(b)}}
