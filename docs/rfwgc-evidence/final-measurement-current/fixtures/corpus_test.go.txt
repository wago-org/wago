func TestFinalPreparationCompile(t *testing.T) {
 for _,m:=range loadCorpus(t) { if !m.supports("Compile") {continue}; t.Run(m.name(),func(t *testing.T){
  mod:=m.decoded(t); if err:=wasm.ValidateModule(mod);err!=nil {t.Fatal(err)}
  cm,err:=benchCompileModule(mod);if err!=nil {t.Fatal(err)};if err=cm.Close();err!=nil {t.Fatal(err)}
  t.Log("FM_LEAF BenchmarkCompile/"+m.name())
 }) }
}
func TestFinalPreparationRuntime(t *testing.T) {
 for _,m:=range loadCorpus(t) {
  if (len(m.Exec)==0 && len(m.SemanticExec)==0)||!m.supports("Exec") {continue}
  c,err:=wago.NewRuntimeConfig().Compile(m.bytes);if err!=nil {t.Fatal(err)};defer c.Close()
  in,err:=wago.Instantiate(c,wago.InstantiateOptions{Imports:hostStubs(c)});if err!=nil {t.Fatal(err)};defer in.Close()
  if m.Init!="" {if _,err=in.Invoke(m.Init);err!=nil {t.Fatal(err)}}
  for _,e:=range m.Exec {t.Run(m.name()+"."+e.Export,func(t *testing.T){
   if e.Want==nil {t.Fatal("missing exact oracle")};args:=make([]uint64,len(e.Args));for i,a:=range e.Args {args[i]=wago.I32(a)}
   fn,err:=in.WasmFunc(e.Export);if err!=nil {t.Fatal(err)}
   got,err:=fn.Invoke(args...);if err!=nil || !slices.Equal(got,e.Want) {t.Fatalf("oracle got=%v want=%v err=%v",got,e.Want,err)}
   if err=wasmFuncInvoker(fn,args)();err!=nil {t.Fatal(err)}
   t.Log("FM_LEAF BenchmarkExec/"+m.name()+"."+e.Export)
  })}
  for _,s:=range semanticExecCases(t,m) {t.Run(m.name()+"."+s.Invoke.Export,func(t *testing.T){
   if err:=runSemanticOracle(s);err!=nil {t.Fatal(err)}
   prepared,err:=prepareWagoSemanticExec(in,s);if err!=nil {t.Fatal(err)};if err=prepared.invoke();err!=nil {t.Fatal(err)}
   t.Log("FM_LEAF BenchmarkExec/"+m.name()+"."+s.Invoke.Export)
  })}
  in.Close();if err=c.Close();err!=nil {t.Fatal(err)}
 }
 for _,m:=range commandCorpus(t) {t.Run(m.name(),func(t *testing.T){
  validateCommandInputs(t,m);stdin:=commandInput(t,m)
  c,err:=wago.Compile(nil,m.bytes);if err!=nil {t.Fatal(err)};defer c.Close()
  got,err:=runWagoCommand(m,c,stdin,true);if err!=nil {t.Fatal(err)};if err=validateCommandOutput(m,got);err!=nil {t.Fatal(err)}
  t.Log("FM_LEAF BenchmarkCommandExec/"+m.name())
 })}
}
