
func TestLoopIntConstHintsSkipShiftMultipliersArm64(t *testing.T) {
	for _, typ := range []uint8{1, 2} {
		width := 32
		op := byte(0x6c)
		if typ == 2 {
			width = 64
			op = 0x7e
		}
		for shift := 0; shift < width; shift++ {
			value := int64(uint64(1) << shift)
			if typ == 1 {
				value = int64(int32(value))
			}
			if loopIntConstNeedsRegister(op, value, typ) {
				t.Fatalf("type=%d shift=%d unnecessarily needs register", typ, shift)
			}
		}
		for _, value := range []int64{0, 3, 7, 17, -1, -3} {
			if !loopIntConstNeedsRegister(op, value, typ) {
				t.Fatalf("type=%d multiplier=%d missing register consumer", typ, value)
			}
		}
	}
}

func TestLoopIntConstShiftMultipliersDoNotCrowdCandidatesArm64(t *testing.T) {
	body := []byte{0x03, 0x40}
	for shift := 0; shift < maxLoopIntConstCandidates; shift++ {
		body = appendI32Const(body, int32(1)<<shift, 0x6c)
	}
	body = appendI32Const(body, 0x12345678, 0x6c)
	body = append(body, 0x0b, 0x0b)
	h, err := scanBodyBytes(body, 0, 0, 0)
	if err != nil {
		t.Fatal(err)
	}
	if h.loopIntConstCount != 1 || h.loopIntConst[0] != 0x12345678 || h.loopIntConstTypes != 1 {
		t.Fatalf("register consumer crowded out: count=%d types=%x constants=%x", h.loopIntConstCount, h.loopIntConstTypes, h.loopIntConst)
	}
}
