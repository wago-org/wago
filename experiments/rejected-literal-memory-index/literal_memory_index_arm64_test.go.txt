//go:build arm64

package arm64

import (
	"testing"

	"github.com/wago-org/wago/src/core/compiler/wasm"
	"github.com/wago-org/wago/tests/support/wasmtest"
)

func literalIndexOptions(on, guard bool, stats *ModuleStats) CompileOptions {
	return CompileOptions{ElideBoundsChecks: guard, Stats: stats, Optimizations: map[string]bool{"literal-memory-index": on}}
}

func literalIndexConst(body []byte, addr uint32) []byte {
	body = append(body, 0x41)
	return append(body, wasmtest.SLEB32(int32(addr))...)
}

func TestLiteralMemoryIndexLoadWidths(t *testing.T) {
	requireCompilerDiagnostics(t)
	cases := []struct {
		op     byte
		size   int
		signed bool
		wide   bool
	}{
		{0x28, 4, false, false}, {0x29, 8, false, true},
		{0x2c, 1, true, false}, {0x2d, 1, false, false},
		{0x2e, 2, true, false}, {0x2f, 2, false, false},
		{0x30, 1, true, true}, {0x31, 1, false, true},
		{0x32, 2, true, true}, {0x33, 2, false, true},
		{0x34, 4, true, true}, {0x35, 4, false, true},
	}
	for _, tc := range cases {
		for _, high := range []bool{false, true} {
			addr, off := uint32(0x2b8), uint32(0)
			if high {
				addr, off = 0x82b8, 8
			}
			body := literalIndexConst([]byte{0}, addr+off)
			body = append(body, 0x20, 0, 0x37, 3, 0) // initialize eight bytes from the i64 argument
			body = literalIndexConst(body, addr)
			body = append(body, tc.op, 0)
			body = append(body, wasmtest.ULEB(off)...)
			body = append(body, 0x0b)
			typ := wasm.I32
			if tc.wide {
				typ = wasm.I64
			}
			m := modMem(t, 1, []wasm.ValType{wasm.I64}, []wasm.ValType{typ}, body)
			for _, guard := range []bool{false, true} {
				for _, on := range []bool{false, true} {
					var stats ModuleStats
					cm, err := CompileModuleWith(m, literalIndexOptions(on, guard, &stats))
					if err != nil {
						t.Fatal(err)
					}
					if cm.CodeImage != nil {
						cm.CodeImage.Close()
					}
					got := stats.Funcs[0].Peephole["literal-memory-index"]
					if on && got != 1 || !on && got != 0 {
						t.Fatalf("op=%x high=%v guard=%v on=%v folds=%d", tc.op, high, guard, on, got)
					}
					for _, v := range []uint64{0, 0x88776655aabbccdd, ^uint64(0)} {
						want := v
						switch tc.size {
						case 1:
							want = uint64(uint8(v))
							if tc.signed {
								want = uint64(int64(int8(v)))
							}
						case 2:
							want = uint64(uint16(v))
							if tc.signed {
								want = uint64(int64(int16(v)))
							}
						case 4:
							want = uint64(uint32(v))
							if tc.signed {
								want = uint64(int64(int32(v)))
							}
						}
						result, err := runArm64WrapperWithOptions(t, m, literalIndexOptions(on, guard, nil), v)
						if !tc.wide {
							result, want = uint64(uint32(result)), uint64(uint32(want))
						}
						if err != nil || result != want {
							t.Fatalf("op=%x high=%v guard=%v on=%v input=%x got=%x want=%x err=%v", tc.op, high, guard, on, v, result, want, err)
						}
					}
				}
			}
		}
	}
}

func TestLiteralMemoryIndexLoadBeforeStore(t *testing.T) {
	requireCompilerDiagnostics(t)
	// The old load must execute before an overwriting store despite having an
	// immutable address with no allocator owner. Then combine it with a new load.
	body := literalIndexConst([]byte{0}, 0x2b8)
	body = append(body, 0x20, 0, 0x37, 3, 0)
	body = literalIndexConst(body, 0x2b8)
	body = append(body, 0x29, 3, 0)
	body = literalIndexConst(body, 0x2b8)
	body = append(body, 0x20, 1, 0x37, 3, 0)
	body = literalIndexConst(body, 0x2b8)
	body = append(body, 0x29, 3, 0, 0x7c, 0x0b)
	m := modMem(t, 1, []wasm.ValType{wasm.I64, wasm.I64}, []wasm.ValType{wasm.I64}, body)
	for _, guard := range []bool{false, true} {
		for _, on := range []bool{false, true} {
			for _, args := range [][2]uint64{{1, 2}, {^uint64(0), 0x8877665544332211}} {
				got, err := runArm64WrapperWithOptions(t, m, literalIndexOptions(on, guard, nil), args[0], args[1])
				if err != nil || got != args[0]+args[1] {
					t.Fatalf("guard=%v on=%v args=%x got=%x err=%v", guard, on, args, got, err)
				}
			}
		}
	}
}

func TestLiteralMemoryIndexLoadTrapsAndWidening(t *testing.T) {
	for _, tc := range []struct{ addr, off uint32 }{{0x102bc, 0}, {65535, 0}, {0xffffffff, 1}, {1, 0x7fffffff}} {
		body := literalIndexConst([]byte{0}, tc.addr)
		body = append(body, 0x28, 2)
		body = append(body, wasmtest.ULEB(tc.off)...)
		body = append(body, 0x1a, 0x41, 7, 0x0b) // dropping the result must still trap
		m := modMem(t, 1, nil, []wasm.ValType{wasm.I32}, body)
		// Signal faults require a genuinely guarded JobMemory; the dedicated
		// guard-page fixture below uses that runtime surface.
		for _, guard := range []bool{false} {
			for _, on := range []bool{false, true} {
				_, err := runArm64WrapperWithOptions(t, m, literalIndexOptions(on, guard, nil))
				if err == nil {
					t.Fatalf("addr=%x off=%x guard=%v on=%v did not trap", tc.addr, tc.off, guard, on)
				}
			}
		}
	}
}

func TestImmutableZeroMemoryAddressOwnership(t *testing.T) {
	if memRefStorage(ZR, 8, 4, false, false, -1, -1).memAddressOwned() {
		t.Fatal("immutable zero index acquired an address owner")
	}
	if memRefStorage(X9, 8, 4, false, false, 3, 3).memAddressOwned() {
		t.Fatal("borrowed local index acquired an address owner")
	}
	if !memRefStorage(X9, 8, 4, false, false, -1, -1).memAddressOwned() {
		t.Fatal("ordinary address lost its owner")
	}
}

func TestLiteralMemoryIndexDeclinesEqualWordCost(t *testing.T) {
	requireCompilerDiagnostics(t)
	// A page-aligned one-word literal + indexed LDR already takes two words.
	body := literalIndexConst([]byte{0}, 0x10000)
	body = append(body, 0x28, 2, 0, 0x0b)
	m := modMem(t, 1, nil, []wasm.ValType{wasm.I32}, body)
	for _, guard := range []bool{false, true} {
		var stats ModuleStats
		cm, err := CompileModuleWith(m, literalIndexOptions(true, guard, &stats))
		if err != nil {
			t.Fatal(err)
		}
		if cm.CodeImage != nil {
			cm.CodeImage.Close()
		}
		if stats.Funcs[0].Peephole["literal-memory-index"] != 0 {
			t.Fatal("equal-word address rewrite admitted")
		}
	}
}

func TestLiteralMemoryIndexLoadPressure(t *testing.T) {
	requireCompilerDiagnostics(t)
	const live = 14
	body := []byte{1, 63, 0x7e}
	for i := 0; i < live; i++ {
		body = append(body, 0x20, 0, 0x20, 1, 0x7c, 0x21, 2, 0x20, 2)
	}
	body = literalIndexConst(body, 0x2b8)
	body = append(body, 0x20, 0, 0x37, 3, 0)
	body = literalIndexConst(body, 0x2b8)
	body = append(body, 0x29, 3, 0, 0x22, 0)
	for i := 0; i < live; i++ {
		body = append(body, 0x7c)
	}
	body = append(body, 0x0b)
	m := modMem(t, 1, []wasm.ValType{wasm.I64, wasm.I64}, []wasm.ValType{wasm.I64}, body)
	for _, guard := range []bool{false, true} {
		for _, on := range []bool{false, true} {
			var stats ModuleStats
			opts := literalIndexOptions(on, guard, &stats)
			opts.Optimizations["interval-region-pins"] = false
			cm, err := CompileModuleWith(m, opts)
			if err != nil {
				t.Fatal(err)
			}
			if cm.CodeImage != nil {
				cm.CodeImage.Close()
			}
			folds := stats.Funcs[0].Peephole["literal-memory-index"]
			if on && folds != 1 || !on && folds != 0 {
				t.Fatalf("guard=%v on=%v folds=%d", guard, on, folds)
			}
			for _, pair := range [][2]uint64{{5, 11}, {^uint64(0), 0x8877665544332211}} {
				opts.Stats = nil
				got, err := runArm64WrapperWithOptions(t, m, opts, pair[0], pair[1])
				want := live*(pair[0]+pair[1]) + pair[0]
				if err != nil || got != want {
					t.Fatalf("guard=%v on=%v args=%x got=%x want=%x err=%v", guard, on, pair, got, want, err)
				}
			}
		}
	}
}
