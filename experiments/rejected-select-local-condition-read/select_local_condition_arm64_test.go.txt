// The wide, loop-free table gives these sources canonical frame homes. Regional
// pinning is disabled explicitly so the fixtures exercise actual load reuse.
func selectConditionOptions(on, guard bool, stats *ModuleStats) CompileOptions {
	return CompileOptions{ElideBoundsChecks: guard, Stats: stats, Optimizations: map[string]bool{
		"select-local-condition-read": on, "interval-region-pins": false,
	}}
}

func selectConditionModule(t *testing.T, wide bool, instructions []byte) *wasm.Module {
	typ := wasm.I32
	localType := byte(0x7f)
	if wide {
		typ = wasm.I64
		localType = 0x7e
	}
	body := []byte{1, 63, localType, 0x3f, 0, 0x1a}
	body = append(body, instructions...)
	body = append(body, 0x0b)
	return modMem(t, 1, []wasm.ValType{typ, typ}, []wasm.ValType{typ}, body)
}

func TestSelectLocalConditionReadWidthsAndAliasing(t *testing.T) {
	requireCompilerDiagnostics(t)
	values := [][2]uint64{{0, 1}, {1, 0}, {0x80000000, 0x7fffffff}, {0xffffffff, 0}, {0x100000001, 2}, {0x8000000000000000, 0}, {^uint64(0), 7}}
	for _, wide := range []bool{false, true} {
		for _, unsigned := range []bool{false, true} {
			lt, add := byte(0x48), byte(0x6a)
			if wide {
				lt, add = 0x53, 0x7c
			}
			if unsigned {
				lt++
			}
			for _, tee := range []bool{false, true} {
				// Preserve an older local0 snapshot below a select which writes local0.
				body := []byte{0x20, 0, 0x20, 0, 0x20, 1, 0x20, 0, 0x20, 1, lt, 0x1b}
				if tee {
					body = append(body, 0x22, 0)
				} else {
					body = append(body, 0x21, 0, 0x20, 0)
				}
				body = append(body, add)
				m := selectConditionModule(t, wide, body)
				for _, guard := range []bool{false, true} {
					for _, on := range []bool{false, true} {
						var stats ModuleStats
						cm, err := CompileModuleWith(m, selectConditionOptions(on, guard, &stats))
						if err != nil {
							t.Fatal(err)
						}
						if cm.CodeImage != nil {
							cm.CodeImage.Close()
						}
						got := stats.Funcs[0].Peephole["select-local-condition-read"]
						if on && got != 2 || !on && got != 0 {
							t.Fatalf("wide=%v unsigned=%v tee=%v guard=%v on=%v loans=%d", wide, unsigned, tee, guard, on, got)
						}
						for _, pair := range values {
							x, y := pair[0], pair[1]
							if !wide {
								x, y = uint64(uint32(x)), uint64(uint32(y))
							}
							less := x < y
							if !unsigned {
								less = int64(x) < int64(y)
								if !wide {
									less = int32(x) < int32(y)
								}
							}
							chosen := y
							if less {
								chosen = x
							}
							want := x + chosen
							result, err := runArm64WrapperWithOptions(t, m, selectConditionOptions(on, guard, nil), x, y)
							if !wide {
								result, want = uint64(uint32(result)), uint64(uint32(want))
							}
							if err != nil || result != want {
								t.Fatalf("wide=%v unsigned=%v tee=%v guard=%v on=%v x=%x y=%x got=%x want=%x err=%v", wide, unsigned, tee, guard, on, x, y, result, want, err)
							}
						}
					}
				}
			}
		}
	}
}

func TestSelectLocalConditionReadVersions(t *testing.T) {
	requireCompilerDiagnostics(t)
	cases := []struct {
		name  string
		body  []byte
		loans int
		want  uint64
	}{
		{"old-selected-source", []byte{0x20, 0, 0x41, 7, 0x21, 0, 0x20, 1, 0x20, 0, 0x45, 0x1b}, 0, 11},
		{"old-condition-source", []byte{0x20, 0, 0x20, 1, 0x20, 0, 0x41, 7, 0x21, 0, 0x45, 0x1b}, 0, 11},
		{"new-condition-source", []byte{0x20, 0, 0x41, 7, 0x21, 0, 0x20, 0, 0x20, 0, 0x45, 0x1b}, 1, 7},
		{"deferred-selected-source", []byte{0x20, 0, 0x41, 1, 0x6a, 0x20, 1, 0x20, 1, 0x20, 0, 0x48, 0x1b}, 1, 11},
		{"same-source-twice", []byte{0x20, 0, 0x20, 0, 0x20, 0, 0x45, 0x1b}, 1, 5},
	}
	for _, tc := range cases {
		t.Run(tc.name, func(t *testing.T) {
			m := selectConditionModule(t, false, tc.body)
			for _, guard := range []bool{false, true} {
				for _, on := range []bool{false, true} {
					var stats ModuleStats
					cm, err := CompileModuleWith(m, selectConditionOptions(on, guard, &stats))
					if err != nil {
						t.Fatal(err)
					}
					if cm.CodeImage != nil {
						cm.CodeImage.Close()
					}
					wantLoans := 0
					if on {
						wantLoans = tc.loans
					}
					if got := stats.Funcs[0].Peephole["select-local-condition-read"]; got != wantLoans {
						t.Fatalf("guard=%v on=%v loans=%d want=%d", guard, on, got, wantLoans)
					}
					got, err := runArm64WrapperWithOptions(t, m, selectConditionOptions(on, guard, nil), 5, 11)
					if err != nil || uint64(uint32(got)) != tc.want {
						t.Fatalf("guard=%v on=%v got=%d want=%d err=%v", guard, on, got, tc.want, err)
					}
				}
			}
		})
	}
}

func TestSelectLocalConditionReadPressure(t *testing.T) {
	requireCompilerDiagnostics(t)
	const live = 14
	// Materialized old values remain live below the selected sources. A deferred
	// RHS then allocates while both selected registers must remain protected.
	body := make([]byte, 0, 180)
	for i := 0; i < live; i++ {
		body = append(body, 0x20, 0, 0x20, 1, 0x6a, 0x21, 2, 0x20, 2)
	}
	body = append(body, 0x20, 0, 0x20, 1, 0x20, 0)
	for i := 0; i < 8; i++ {
		body = append(body, 0x20, 1)
	}
	for i := 0; i < 7; i++ {
		body = append(body, 0x6a)
	}
	body = append(body, 0x48, 0x1b, 0x22, 0)
	for i := 0; i < live; i++ {
		body = append(body, 0x6a)
	}
	m := selectConditionModule(t, false, body)
	for _, guard := range []bool{false, true} {
		for _, on := range []bool{false, true} {
			var stats ModuleStats
			cm, err := CompileModuleWith(m, selectConditionOptions(on, guard, &stats))
			if err != nil {
				t.Fatal(err)
			}
			if cm.CodeImage != nil {
				cm.CodeImage.Close()
			}
			wantLoans := 0
			if on {
				wantLoans = 1
			}
			if got := stats.Funcs[0].Peephole["select-local-condition-read"]; got != wantLoans {
				t.Fatalf("guard=%v on=%v loans=%d want=%d", guard, on, got, wantLoans)
			}
			for _, pair := range [][2]uint64{{5, 11}, {100, 1}, {0x80000000, 0xffffffff}} {
				x, y := uint32(pair[0]), uint32(pair[1])
				chosen := y
				if int32(x) < int32(8*y) {
					chosen = x
				}
				want := live*(x+y) + chosen
				got, err := runArm64WrapperWithOptions(t, m, selectConditionOptions(on, guard, nil), pair[0], pair[1])
				if err != nil || uint32(got) != want {
					t.Fatalf("guard=%v on=%v pair=%x got=%x want=%x err=%v", guard, on, pair, got, want, err)
				}
			}
		}
	}
}

func TestMatchingSelectedLocalReadRequiresCurrentIntegerType(t *testing.T) {
	selected := &elem{st: storage{kind: stLocalRef, typ: mtI32, idx: 7}}
	for _, st := range []storage{
		{kind: stLocalRef, typ: mtI64, idx: 7},
		{kind: stLocalRef, typ: mtI32, idx: 8},
		{kind: stLocalRef, typ: mtF32, idx: 7},
		{kind: stReg, typ: mtI32, idx: 7},
		{kind: stSlot, typ: mtI32, idx: 7},
	} {
		if matchingSelectedLocalRead(&elem{st: st}, selected) {
			t.Fatalf("unexpected match: %+v", st)
		}
	}
}
