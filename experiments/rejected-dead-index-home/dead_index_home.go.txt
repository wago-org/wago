//go:build arm64

package arm64

import (
	"os"

	"github.com/wago-org/wago/src/core/compiler/wasm"
)

var deadIndexHomeEnabled = os.Getenv("WAGO_ARM64_EXPERIMENT_DEAD_INDEX_HOME") == "1"

const (
	maxDeadIndexBody  = 4 << 10
	maxDeadIndexNodes = 1024
	maxDeadIndexEdges = 2048
	noDeadIndexEdge   = ^uint16(0)
)

// The graph exists only in worker scratch, and only after a simple borrowed
// branch-index candidate appears. Local masks describe value liveness, not
// register placement; a definition kills the preceding value on that path.
type deadIndexNode struct {
	live              uint64
	pos, first, count uint16
	local, access     uint8
}

type deadIndexEdge struct{ to, link uint16 }

type deadIndexFrame struct {
	kind                      byte
	start, pending, falseEdge uint16
}

// tryDeadIndexHome consumes a non-trapping expression into a permanent local
// home only if the incoming local value is dead on every dispatch successor.
// The mapping itself never changes. Older operand-stack snapshots are copied
// before the destructive write, including deferred loads and trapping trees.
func (f *fn) tryDeadIndexHome(index *elem) bool {
	if len(f.deadIndexBody) == 0 || f.inlineDepth != 0 || f.localBase != 0 || index == nil ||
		!index.isDeferred() || !isBinALU(index.deferredOp()) {
		return false
	}
	left, right := f.s.arg0(index), f.s.arg1(index)
	if left == nil || right == nil || left.elemKind() != ekValue || left.st.kind != stLocalReg ||
		right.elemKind() != ekValue || right.st.kind != stConst || left.st.typ.isXMM() {
		return false
	}
	x, reg := left.st.index(), left.st.reg
	if x < 0 || x >= 64 || f.regUser[reg] != nil ||
		len(f.intervalLast) != 0 && f.intervalOwner[reg] >= 0 {
		return false
	}
	if !f.deadIndexReady && !f.deadIndexFailed {
		f.tmpDeadIndexNodes, f.tmpDeadIndexEdges, f.deadIndexReady = buildDeadIndexGraph(
			f.deadIndexBody, f.classifier, f.tmpDeadIndexNodes, f.tmpDeadIndexEdges)
		f.deadIndexFailed = !f.deadIndexReady
	}
	if !f.deadIndexReady {
		return false
	}
	pc := uint32(f.wasmPC - f.tracePCBase)
	for _, node := range f.tmpDeadIndexNodes {
		if uint32(node.pos) != pc {
			continue
		}
		var live uint64
		for _, edge := range f.tmpDeadIndexEdges[node.first : node.first+node.count] {
			if int(edge.to) < len(f.tmpDeadIndexNodes) {
				live |= f.tmpDeadIndexNodes[edge.to].live
			}
		}
		if live&(uint64(1)<<uint(x)) != 0 {
			return false
		}
		f.realizeLocalRefs(x, f.s.baseOfValentBlock(index))
		f.condenseInto(index, reg)
		f.stats.peep("dead-index-home")
		return true
	}
	return false
}

// buildDeadIndexGraph decodes structured control once. Forward edges are linked
// through their target frame until its end is encountered; loop backedges use
// the known source header. Unsupported control, malformed input, and bounded
// storage/fixed-point exhaustion fail closed.
func buildDeadIndexGraph(body []byte, classifier wasm.ModuleInstructionClassifier, nodes []deadIndexNode, edges []deadIndexEdge) ([]deadIndexNode, []deadIndexEdge, bool) {
	nodes, edges = nodes[:0], edges[:0]
	if len(body) == 0 || len(body) > maxDeadIndexBody {
		return nodes, edges, false
	}
	var frames [64]deadIndexFrame
	frames[0] = deadIndexFrame{kind: 0, pending: noDeadIndexEdge, falseEdge: noDeadIndexEdge}
	depth := 1
	r := wasm.ReaderFrom(body)
	appendEdge := func(to uint16) uint16 {
		idx := uint16(len(edges))
		edges = append(edges, deadIndexEdge{to: to, link: noDeadIndexEdge})
		return idx
	}
	branchEdge := func(label uint32) bool {
		if uint64(label) >= uint64(depth) || len(edges) == maxDeadIndexEdges {
			return false
		}
		fr := &frames[depth-1-int(label)]
		if fr.kind == 0 {
			appendEdge(noDeadIndexEdge)
			return true
		}
		if fr.kind == 0x03 {
			appendEdge(fr.start)
			return true
		}
		e := appendEdge(noDeadIndexEdge)
		edges[e].link, fr.pending = fr.pending, e
		return true
	}
	for r.HasNext() && depth != 0 {
		if len(nodes) == maxDeadIndexNodes || len(edges)+2 > maxDeadIndexEdges {
			return nodes, edges, false
		}
		pos := r.Offset()
		op, err := r.Byte()
		if err != nil {
			return nodes, edges, false
		}
		i := uint16(len(nodes))
		node := deadIndexNode{pos: uint16(pos), first: uint16(len(edges))}
		canFallThrough := true
		switch op {
		case 0x00, 0x0f: // unreachable / return
			canFallThrough = false
		case 0x12, 0x13, 0x15: // tail calls never return to this caller's locals
			var imm wasm.InstructionImmediate
			if err := classifier.ClassifyInto(&r, op, &imm); err != nil {
				return nodes, edges, false
			}
			canFallThrough = false
		case 0x02, 0x03, 0x04: // block / loop / if
			if depth == len(frames) {
				return nodes, edges, false
			}
			if err := wasm.SkipInstructionImmediate(&r, op); err != nil {
				return nodes, edges, false
			}
			fr := deadIndexFrame{kind: op, start: i + 1, pending: noDeadIndexEdge, falseEdge: noDeadIndexEdge}
			if op == 0x04 {
				fr.falseEdge = appendEdge(noDeadIndexEdge)
			}
			frames[depth] = fr
			depth++
		case 0x05: // else: false arm starts here; true arm skips to end
			fr := &frames[depth-1]
			if fr.kind != 0x04 || fr.falseEdge == noDeadIndexEdge {
				return nodes, edges, false
			}
			edges[fr.falseEdge].to = i + 1
			fr.falseEdge = noDeadIndexEdge
			if !branchEdge(0) {
				return nodes, edges, false
			}
			canFallThrough = false
		case 0x0b: // end
			fr := &frames[depth-1]
			for edge := fr.pending; edge != noDeadIndexEdge; edge = edges[edge].link {
				edges[edge].to = i + 1
			}
			if fr.falseEdge != noDeadIndexEdge {
				edges[fr.falseEdge].to = i + 1
			}
			depth--
			if depth == 0 {
				canFallThrough = false
			}
		case 0x0c, 0x0d: // br / br_if
			label, err := r.U32()
			if err != nil || !branchEdge(label) {
				return nodes, edges, false
			}
			canFallThrough = op == 0x0d
		case 0x0e: // br_table
			n, err := r.U32()
			if err != nil || uint64(n)+1 > uint64(maxDeadIndexEdges-len(edges)) {
				return nodes, edges, false
			}
			for j := uint32(0); j <= n; j++ {
				label, err := r.U32()
				if err != nil || !branchEdge(label) {
					return nodes, edges, false
				}
			}
			canFallThrough = false
		case 0x20, 0x21, 0x22:
			x, err := r.U32()
			if err != nil || x >= 64 {
				return nodes, edges, false
			}
			node.local, node.access = uint8(x), op
		case 0x06, 0x07, 0x08, 0x09, 0x0a, 0x18, 0x19, 0x1f:
			return nodes, edges, false // EH can transfer to an unmodeled handler
		default:
			if op >= 0xd0 {
				return nodes, edges, false
			} // extended control / GC / helper instructions
			// Pure arithmetic, memory effects, and ordinary calls do not access
			// private caller locals. Their argument local.get instructions are
			// already explicit nodes. Traps can only terminate a path, so retaining
			// fall-through conservatively is sufficient for this dead-value proof.
			if _, ok := wasm.ImmediateFreeInstructionKind(op); ok {
				continue
			}
			var imm wasm.InstructionImmediate
			if err := classifier.ClassifyInto(&r, op, &imm); err != nil {
				return nodes, edges, false
			}
			continue
		}
		if canFallThrough {
			appendEdge(i + 1)
		}
		node.count = uint16(len(edges)) - node.first
		nodes = append(nodes, node)
	}
	if depth != 0 || r.HasNext() {
		return nodes, edges, false
	}
	// Backward monotone dataflow. Every bit changes from dead to live at most
	// once; a bounded pass budget handles nested cycles conservatively.
	for pass := 0; pass < 64; pass++ {
		changed := false
		for i := len(nodes) - 1; i >= 0; i-- {
			node := &nodes[i]
			var live uint64
			for _, edge := range edges[node.first : node.first+node.count] {
				if int(edge.to) < len(nodes) {
					live |= nodes[edge.to].live
				}
			}
			if node.access == 0x20 {
				live |= uint64(1) << node.local
			} else if node.access != 0 {
				live &^= uint64(1) << node.local
			}
			if live != node.live {
				node.live = live
				changed = true
			}
		}
		if !changed {
			return nodes, edges, true
		}
	}
	return nodes, edges, false
}
