//go:build arm64

package arm64

import "os"

var entryDeadZerosEnabled = os.Getenv("WAGO_ARM64_EXPERIMENT_ENTRY_DEAD_ZEROS") == "1"

// Restrict the analysis to call-making integer functions with enough eager
// initial homes to repay a structured scan. Call-free functions already retain
// locals cheaply; references and EH need independent root/handler proofs.
func (f *fn) tryZeroDeadIntegerLocals(localScores []uint32) bool {
	if !f.makesCalls || f.lazyZero || f.m == nil || f.ft == nil || f.inlineDepth != 0 || f.localBase != 0 ||
		len(f.inlineBase) != 0 || len(f.localType) != f.nLocals || f.moduleEH || len(f.gcTypeLayouts) != 0 ||
		f.gcFrameRoots != nil || len(f.customInstructions) != 0 {
		return false
	}
	idx := int(f.traceFuncIdx) - f.m.ImportedFuncCount()
	if idx < 0 || idx >= len(f.m.Code) {
		return false
	}
	body := f.m.Code[idx].BodyBytes
	if f.nLocals > 64 || f.nLocals-f.nParams < 8 || len(body) == 0 || len(body) > maxLocalInitializationBody ||
		!sigIsIntOnly(f.ft) {
		return false
	}
	for _, typ := range f.localType {
		if typ != mtI32 && typ != mtI64 {
			return false
		}
	}
	var candidates uint64
	for x := f.nParams; x < f.nLocals; x++ {
		if f.opt(optEntryInitElision) && x < len(localScores) && localScores[x]&localScoreEntryInitialized != 0 {
			continue // the existing prefix proof already eliminates this zero
		}
		candidates |= uint64(1) << uint(x)
	}
	if candidates == 0 {
		return false
	}
	needed, ok := scanInitialLocalReads(body, f.classifier)
	if !ok {
		return false
	}
	if mask := candidates &^ needed; mask != 0 {
		f.stats.peep("entry-dead-zeros-analysis")
		f.zeroDeadIntegerLocals(localScores, mask)
		return true
	}
	return false
}

// Eligible functions have only integer locals and no EH/GC lane initialization.
// Select this cold path once; the ordinary OFF/unaffected eager-zero loop retains
// its original work per local. Scores remain immutable, and prefix and new-proof
// counters retain their independent policy meaning.
func (f *fn) zeroDeadIntegerLocals(localScores []uint32, dead uint64) {
	pairZeros := f.opt(optEntryZeroPairs)
	prefixProof := f.opt(optEntryInitElision)
	pendingZero := int32(-1)
	for x := f.nParams; x < f.nLocals; x++ {
		prefix := prefixProof && x < len(localScores) && localScores[x]&localScoreEntryInitialized != 0
		if prefix || dead&(uint64(1)<<uint(x)) != 0 {
			if pairZeros {
				pendingZero = f.flushDeclaredZeroSlot(pendingZero)
			}
			if prefix {
				f.stats.peep("entry-init-elide")
			} else {
				f.stats.peep("entry-dead-zero-elide")
			}
			continue
		}
		if reg, _, ok := f.pinReg(x); ok {
			if pairZeros {
				pendingZero = f.flushDeclaredZeroSlot(pendingZero)
			}
			f.a.MovImm64(reg, 0)
		} else if pairZeros {
			pendingZero = f.queueDeclaredZeroSlot(pendingZero, f.localOff(x))
		} else {
			f.st64(SP, f.localOff(x), ZR)
		}
	}
	if pairZeros {
		f.flushDeclaredZeroSlot(pendingZero)
	}
}
