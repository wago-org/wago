func TestInitialLocalReadsModuleReference(t *testing.T) {
	path := os.Getenv("WAGO_LOCAL_INIT_PROOF_MODULE")
	if path == "" {
		t.Skip("optional source-proof audit")
	}
	data, err := os.ReadFile(path)
	if err != nil {
		t.Fatal(err)
	}
	m, err := wasm.DecodeModule(data)
	if err != nil {
		t.Fatal(err)
	}
	classifier := wasm.NewModuleInstructionClassifier(m, true)
	checked := 0
	for i, c := range m.Code {
		needed, ok := scanInitialLocalReads(c.BodyBytes, classifier)
		nodes, _, referenceOK := buildStructuredLocalLiveness(c.BodyBytes, classifier, nil, nil)
		if !ok || !referenceOK {
			continue
		}
		if needed != nodes[0].live {
			t.Fatalf("function%d one-pass=%x graph=%x", i, needed, nodes[0].live)
		}
		checked++
	}
	if checked == 0 {
		t.Fatal("no bounded functions checked")
	}
	t.Logf("one-pass initial-value proof matches reference CFG for%dfunctions", checked)
}

func TestStructuredLocalLiveness(t *testing.T) {
	for _, tc := range []struct {
		name string
		body []byte
		pc   int
		dead bool
	}{
		{"overwrite-both-targets", []byte{0x02, 0x40, 0x02, 0x40, 0x20, 0, 0x0e, 1, 0, 1, 0x0b, 0x41, 11, 0x21, 0, 0x0f, 0x0b, 0x41, 22, 0x21, 0, 0x20, 0, 0x0b}, 6, true},
		{"one-live-target", []byte{0x02, 0x40, 0x02, 0x40, 0x20, 0, 0x0e, 1, 0, 1, 0x0b, 0x41, 11, 0x21, 0, 0x0f, 0x0b, 0x20, 0, 0x0b}, 6, false},
		{"backedge-overwrite", []byte{0x03, 0x40, 0x41, 7, 0x21, 0, 0x02, 0x40, 0x20, 0, 0x0e, 1, 0, 1, 0x0b, 0x0c, 0, 0x0b, 0x0b}, 10, true},
		{"backedge-read", []byte{0x03, 0x40, 0x20, 0, 0x1a, 0x02, 0x40, 0x20, 0, 0x0e, 1, 0, 1, 0x0b, 0x0c, 0, 0x0b, 0x0b}, 9, false},
		{"if-one-live-arm", []byte{0x02, 0x40, 0x20, 0, 0x0e, 0, 0, 0x0b, 0x41, 1, 0x04, 0x40, 0x41, 7, 0x21, 0, 0x05, 0x20, 0, 0x1a, 0x0b, 0x0b}, 4, false},
		{"if-both-dead-arms", []byte{0x02, 0x40, 0x20, 0, 0x0e, 0, 0, 0x0b, 0x41, 1, 0x04, 0x40, 0x41, 7, 0x21, 0, 0x05, 0x41, 8, 0x21, 0, 0x0b, 0x20, 0, 0x0b}, 4, true},
	} {
		t.Run(tc.name, func(t *testing.T) {
			nodes, edges, ok := buildStructuredLocalLiveness(tc.body, wasm.NewModuleInstructionClassifier(nil, false), nil, nil)
			if !ok {
				t.Fatal("graph rejected")
			}
			needed, scanOK := scanInitialLocalReads(tc.body, wasm.NewModuleInstructionClassifier(nil, false))
			if !scanOK || needed != nodes[0].live {
				t.Fatalf("one-pass=%x ok=%v graph=%x", needed, scanOK, nodes[0].live)
			}
			for _, node := range nodes {
				if int(node.pos) != tc.pc {
					continue
				}
				var live uint64
				for _, edge := range edges[node.first : node.first+node.count] {
					if int(edge.to) < len(nodes) {
						live |= nodes[edge.to].live
					}
				}
				if got := live&1 == 0; got != tc.dead {
					t.Fatalf("dead=%v want%v", got, tc.dead)
				}
				return
			}
			t.Fatal("branch node missing")
		})
	}
}

func TestStructuredEntryLocalLivenessAcrossCalls(t *testing.T) {
	for _, tc := range []struct {
		name string
		body []byte
		want uint64
	}{
		{"overwritten-after-call", []byte{0x10, 0, 0x41, 7, 0x21, 0, 0x20, 0, 0x0b}, 0},
		{"call-argument-reads-zero", []byte{0x20, 0, 0x10, 0, 0x0b}, 1},
		{"live-across-call", []byte{0x10, 0, 0x20, 0, 0x0b}, 1},
		{"tail-call", []byte{0x12, 0, 0x0b}, 0},
		{"bulk-memory-before-overwrite", []byte{0xfc, 10, 0, 0, 0x41, 7, 0x21, 0, 0x20, 0, 0x0b}, 0},
		{"bulk-memory-input-reads-zero", []byte{0x20, 0, 0xfc, 11, 0, 0x0b}, 1},
		{"typed-call-argument-reads-zero", []byte{0x20, 0, 0x25, 0, 0x14, 0, 0x0b}, 1},
		{"typed-tail-call-argument-reads-zero", []byte{0x20, 0, 0x25, 0, 0x15, 0, 0x0b}, 1},
	} {
		t.Run(tc.name, func(t *testing.T) {
			nodes, _, ok := buildStructuredLocalLiveness(tc.body, wasm.NewModuleInstructionClassifier(nil, false), nil, nil)
			needed, scanOK := scanInitialLocalReads(tc.body, wasm.NewModuleInstructionClassifier(nil, false))
			if !scanOK || needed != tc.want {
				t.Fatalf("one-pass=%x ok=%v want=%x", needed, scanOK, tc.want)
			}
			if !ok || nodes[0].live != tc.want {
				t.Fatalf("ok=%v nodes=%v want=%x", ok, nodes, tc.want)
			}
		})
	}
}

func TestInitialLocalReadsNestedEdges(t *testing.T) {
	for _, tc := range []struct {
		name string
		body []byte
		want uint64
	}{
		{"outer-block-skips-definition", []byte{0x02, 0x40, 0x03, 0x40, 0x02, 0x40, 0x03, 0x40, 0x20, 0, 0x0d, 3, 0x41, 7, 0x21, 1, 0x20, 0, 0x0d, 1, 0x0c, 2, 0x0b, 0x0b, 0x20, 1, 0x1a, 0x0c, 0, 0x0b, 0x0b, 0x20, 1, 0x1a, 0x0b}, 3},
		{"outer-loop-backedge-cannot-clear-write", []byte{0x41, 7, 0x21, 1, 0x02, 0x40, 0x03, 0x40, 0x02, 0x40, 0x03, 0x40, 0x20, 0, 0x0d, 3, 0x20, 1, 0x1a, 0x20, 0, 0x0d, 1, 0x0c, 2, 0x0b, 0x0b, 0x20, 1, 0x1a, 0x0c, 0, 0x0b, 0x0b, 0x20, 1, 0x1a, 0x0b}, 1},
		{"backedge-before-write-and-forward-exit-after-write", []byte{0x03, 0x40, 0x02, 0x40, 0x03, 0x40, 0x20, 0, 0x0d, 2, 0x41, 7, 0x21, 1, 0x0c, 1, 0x0b, 0x0b, 0x20, 1, 0x1a, 0x0b, 0x0b}, 1},
		{"branch-table-outer-block-and-loop", []byte{0x02, 0x40, 0x03, 0x40, 0x02, 0x40, 0x20, 0, 0x0e, 2, 0, 1, 2, 0x0b, 0x41, 7, 0x21, 1, 0x0c, 1, 0x0b, 0x0b, 0x20, 1, 0x1a, 0x0b}, 3},
		{"unreachable-then-defined-else", []byte{0x20, 0, 0x04, 0x40, 0x00, 0x05, 0x41, 7, 0x21, 1, 0x0b, 0x20, 1, 0x1a, 0x0b}, 1},
		{"function-label-then-live-else", []byte{0x20, 0, 0x04, 0x40, 0x0c, 1, 0x05, 0x20, 1, 0x1a, 0x0b, 0x0b}, 3},
		{"function-label-does-not-revive-dead-tail", []byte{0x41, 7, 0x21, 1, 0x20, 1, 0x1a, 0x0c, 0, 0x20, 2, 0x1a, 0x0b}, 0},
		{"function-branch-table-terminal", []byte{0x20, 0, 0x0e, 1, 0, 0, 0x20, 1, 0x1a, 0x0b}, 1},
	} {
		t.Run(tc.name, func(t *testing.T) {
			classifier := wasm.NewModuleInstructionClassifier(nil, false)
			got, ok := scanInitialLocalReads(tc.body, classifier)
			nodes, _, referenceOK := buildStructuredLocalLiveness(tc.body, classifier, nil, nil)
			if !ok || got != tc.want || !referenceOK || got != nodes[0].live {
				t.Fatalf("scan=%x ok=%v want=%x graph=%v referenceOK=%v", got, ok, tc.want, nodes, referenceOK)
			}
		})
	}
}

func TestInitialLocalReadsRejectsUnmodeledOrMalformedControl(t *testing.T) {
	deep := []byte{}
	for i := 0; i < 64; i++ {
		deep = append(deep, 0x02, 0x40)
	}
	for i := 0; i < 65; i++ {
		deep = append(deep, 0x0b)
	}
	for _, body := range [][]byte{nil, {0x05, 0x0b}, {0x0c, 1, 0x0b}, {0x0e, 0xff, 0xff, 0xff, 0xff, 0x0f, 0x0b}, {0x20, 64, 0x0b}, {0x06, 0x40, 0x0b, 0x0b}, {0xfb, 0, 0x0b}, {0xfe, 0, 0x0b}, {0xff, 0x0b}, deep} {
		if _, ok := scanInitialLocalReads(body, wasm.NewModuleInstructionClassifier(nil, false)); ok {
			t.Fatalf("unexpected admission of%x", body)
		}
	}
}
