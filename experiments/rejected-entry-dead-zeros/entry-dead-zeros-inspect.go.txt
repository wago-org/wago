//go:build ignore

package main

import (
	"encoding/json"
	"os"

	"github.com/wago-org/wago/src/core/compiler/backend/railshot/arm64"
	"github.com/wago-org/wago/src/core/compiler/wasm"
)

func main() {
	data, err := os.ReadFile(os.Args[1])
	if err != nil {
		panic(err)
	}
	m, err := wasm.DecodeModule(data)
	if err != nil {
		panic(err)
	}
	enc := json.NewEncoder(os.Stdout)
	for _, enabled := range []bool{false, true} {
		stats := &arm64.ModuleStats{}
		cm, err := arm64.CompileModuleWith(m, arm64.CompileOptions{Stats: stats, Workers: 1, ElideBoundsChecks: true, Optimizations: map[string]bool{"entry-dead-zeros": enabled, "inline": os.Getenv("ENTRY_INSPECT_NO_INLINE") != "1", "inline-callfree": os.Getenv("ENTRY_INSPECT_NO_INLINE") != "1"}})
		if err != nil {
			panic(err)
		}
		for _, f := range stats.Funcs {
			if f.Peephole["entry-dead-zeros-analysis"] == 0 && f.FuncIdx != 19 {
				continue
			}
			declared := 0
			for _, run := range m.Code[f.FuncIdx].Locals.Runs {
				declared += int(run.Count)
			}
			if err := enc.Encode(map[string]any{"on": enabled, "function": f.FuncIdx, "declared": declared, "body": len(m.Code[f.FuncIdx].BodyBytes), "bytes": f.CodeBytes, "initial_prefix_elisions": f.Peephole["entry-init-elide"], "incremental_elisions": f.Peephole["entry-dead-zero-elide"], "peephole": f.Peephole}); err != nil {
				panic(err)
			}
		}
		if cm.CodeImage != nil {
			cm.CodeImage.Close()
		}
	}
}
